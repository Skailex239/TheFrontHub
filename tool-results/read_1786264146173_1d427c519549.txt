   246→   246→export type UpdateUsernameResult =
   247→   247→  | { ok: true; data: PutUsernameResponse }
   248→   248→  | { ok: false; code: "invalid"; message?: string }
   249→   249→  | { ok: false; code: "profane" }
   250→   250→  | { ok: false; code: "taken" }
   251→   251→  | { ok: false; code: "cooldown"; retryAfterSeconds: number | null }
   252→   252→  | { ok: false; code: "failed" };
   253→   253→
   254→   254→// PUT /users/@me/username — renames the account username. Every failure is
   255→   255→// atomic (no name change, no cooldown consumed). Both 409 bodies ("name
   256→   256→// exclusively held" and "suffix space exhausted") map to "taken": the user
   257→   257→// remedy is the same — pick another name. Invalidates the cached /users/@me
   258→   258→// on success so the next read reflects the new name.
   259→   259→export async function updateUsername(
   260→   260→  username: string,
   261→   261→): Promise<UpdateUsernameResult> {
   262→   262→  try {
   263→   263→    const response = await fetch(`${getApiBase()}/users/@me/username`, {
   264→   264→      method: "PUT",
   265→   265→      headers: {
   266→   266→        "Content-Type": "application/json",
   267→   267→        Authorization: await getAuthHeader(),
   268→   268→      },
   269→   269→      body: JSON.stringify({ username }),
   270→   270→    });
   271→   271→    if (response.status === 401) {
   272→   272→      await logOut();
   273→   273→      return { ok: false, code: "failed" };
   274→   274→    }
   275→   275→    if (response.status === 400) {
   276→   276→      const body = await response.json().catch(() => null);
   277→   277→      if (body?.code === "USERNAME_PROFANE") {
   278→   278→        return { ok: false, code: "profane" };
   279→   279→      }
   280→   280→      return {
   281→   281→        ok: false,
   282→   282→        code: "invalid",
   283→   283→        message: typeof body?.reason === "string" ? body.reason : undefined,
   284→   284→      };
   285→   285→    }
   286→   286→    if (response.status === 409) {
   287→   287→      return { ok: false, code: "taken" };
   288→   288→    }
   289→   289→    if (response.status === 429) {
   290→   290→      const retryAfter = response.headers.get("Retry-After");
   291→   291→      const seconds = retryAfter === null ? NaN : Number(retryAfter);
   292→   292→      return {
   293→   293→        ok: false,
   294→   294→        code: "cooldown",
   295→   295→        retryAfterSeconds: Number.isFinite(seconds) ? seconds : null,
   296→   296→      };
   297→   297→    }
   298→   298→    if (!response.ok) {
   299→   299→      console.error(
   300→   300→        "updateUsername: request failed",
   301→   301→        response.status,
   302→   302→        response.statusText,
   303→   303→      );
   304→   304→      return { ok: false, code: "failed" };
   305→   305→    }
   306→   306→    const parsed = PutUsernameResponseSchema.safeParse(await response.json());
   307→   307→    if (!parsed.success) {
   308→   308→      console.error("updateUsername: Zod validation failed", parsed.error);
   309→   309→      return { ok: false, code: "failed" };
   310→   310→    }
   311→   311→    invalidateUserMe();
   312→   312→    return { ok: true, data: parsed.data };
   313→   313→  } catch (e) {
   314→   314→    console.error("updateUsername: request failed", e);
   315→   315→    return { ok: false, code: "failed" };
   316→   316→  }
   317→   317→}
