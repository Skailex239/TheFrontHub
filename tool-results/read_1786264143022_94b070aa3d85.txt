   185→      });
   186→      if (response.status === 401) {
   187→        await logOut();
   188→        return false;
   189→      }

   197→      }
   198→      return result.data;
   199→    } catch (e) {
   200→      return false;
   201→    }

   246→export type UpdateUsernameResult =
   247→  | { ok: true; data: PutUsernameResponse }
   248→  | { ok: false; code: "invalid"; message?: string }
   249→  | { ok: false; code: "profane" }
   250→  | { ok: false; code: "taken" }
   251→  | { ok: false; code: "cooldown"; retryAfterSeconds: number | null }
   252→  | { ok: false; code: "failed" };
   253→
   254→// PUT /users/@me/username — renames the account username. Every failure is
   255→// atomic (no name change, no cooldown consumed). Both 409 bodies ("name
   256→// exclusively held" and "suffix space exhausted") map to "taken": the user
   257→// remedy is the same — pick another name. Invalidates the cached /users/@me
   258→// on success so the next read reflects the new name.
   259→export async function updateUsername(
   260→  username: string,
   261→): Promise<UpdateUsernameResult> {
   262→  try {
   263→    const response = await fetch(`${getApiBase()}/users/@me/username`, {
   264→      method: "PUT",
   265→      headers: {
   266→        "Content-Type": "application/json",
   267→        Authorization: await getAuthHeader(),
   268→      },
   269→      body: JSON.stringify({ username }),
   270→    });
   271→    if (response.status === 401) {
   272→      await logOut();
   273→      return { ok: false, code: "failed" };
   274→    }
   275→    if (response.status === 400) {
   276→      const body = await response.json().catch(() => null);
   277→      if (body?.code === "USERNAME_PROFANE") {
   278→        return { ok: false, code: "profane" };
   279→      }
   280→      return {
   281→        ok: false,
   282→        code: "invalid",
   283→        message: typeof body?.reason === "string" ? body.reason : undefined,
   284→      };
   285→    }
   286→    if (response.status === 409) {
   287→      return { ok: false, code: "taken" };
   288→    }
   289→    if (response.status === 429) {
   290→      const retryAfter = response.headers.get("Retry-After");
   291→      const seconds = retryAfter === null ? NaN : Number(retryAfter);
   292→      return {
   293→        ok: false,
   294→        code: "cooldown",
   295→        retryAfterSeconds: Number.isFinite(seconds) ? seconds : null,
   296→      };
   297→    }
   298→    if (!response.ok) {
   299→      console.error(
   300→        "updateUsername: request failed",
   301→        response.status,
   302→        response.statusText,
   303→      );
   304→      return { ok: false, code: "failed" };
   305→    }
   306→    const parsed = PutUsernameResponseSchema.safeParse(await response.json());
   307→    if (!parsed.success) {
   308→      console.error("updateUsername: Zod validation failed", parsed.error);
   309→      return { ok: false, code: "failed" };
   310→    }
   311→    invalidateUserMe();
   312→    return { ok: true, data: parsed.data };
   313→  } catch (e) {
   314→    console.error("updateUsername: request failed", e);
   315→    return { ok: false, code: "failed" };
   316→  }
   317→}

   439→// POST /users/@me/tribe_names/:id/boosts — buy a 30-day boost for an active
   440→// tribe name. Boosts stack without limit and the endpoint has no rate limit,
   441→// so the Idempotency-Key is what stops a double-submit from charging twice:
   442→// callers generate a fresh UUID per user-initiated click. Nothing is charged
   443→// on any error. Spends hard currency — callers refresh /users/@me and the
   444→// name list afterwards (boost state is only served by the list endpoint).
   445→export async function boostTribeName(
   446→  id: string,
   447→  idempotencyKey: string,
   448→): Promise<BoostTribeNameResult> {
   449→  try {

   451→      `${getApiBase()}/users/@me/tribe_names/${encodeURIComponent(id)}/boosts`,
   452→      {
   453→        method: "POST",
   454→        headers: {
   455→          Authorization: await getAuthHeader(),
   456→          "Idempotency-Key": idempotencyKey,
   457→        },
   458→      },
   459→    );
   460→    if (response.status === 401) {
   461→      await logOut();
   462→      return { ok: false, code: "failed" };
   463→    }
   464→    if (response.status === 400) {
   465→      const body = await response.json().catch(() => null);

   473→      return { ok: false, code: "failed" };
   474→    }

   476→      return { ok: false, code: "not_found" };
   477→    }
   478→    if (!response.ok) {
   479→      console.error(
   480→        "boostTribeName: request failed",

   482→        response.statusText,

  1064→// signal that a second one exists — and the common case early on is one

  1067→  TribeLeaderboardResponse | false

  1072→

  1075→  if (second === false) return first;
  1076→

  1102→// Fetch an API-served JSON config (news.json-style: served file + bundled fallback).
  1103→// Any error, non-200, or invalid payload falls back, parsed through the same schema so
  1104→// the fallback is validated too. The timeout guards recurring callers (StreamsFeed polls
  1105→// every 60s) from a fetch that never settles.
  1106→async function getServedConfig<T>(
  1107→  name: string,
  1108→  schema: z.ZodType<T>,
  1109→  fallback: unknown,
  1110→): Promise<T> {
  1111→  try {
  1112→    const res = await fetch(`${getApiBase()}/${name}`, {
  1113→      headers: { Accept: "application/json" },
  1114→      signal: AbortSignal.timeout(10_000),
  1115→    });
  1116→    if (res.status !== 200) {
  1117→      console.warn(`${name}: unexpected status`, res.status);
  1118→      return schema.parse(fallback);
  1119→    }
  1120→    const parsed = schema.safeParse(await res.json());
  1121→    if (!parsed.success) {
  1122→      console.warn(`${name}: Zod validation failed`, parsed.error);
  1123→      return schema.parse(fallback);
  1124→    }
  1125→    return parsed.data;
  1126→  } catch (err) {
  1127→    console.warn(`${name}: request failed, using fallback`, err);
  1128→    return schema.parse(fallback);
  1129→  }
  1130→}
