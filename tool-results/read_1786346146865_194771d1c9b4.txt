    31→    31→// VIP skin: publicId → rewardType (matching par PUBLIC ID, pas par alias)
    32→    32→let vipPlayersByPid = new Map();
    33→    33→let _vipUnsub = null;
    34→    34→const NEW_SKIN_TYPES = ['cyberpunk','sunset','aurore','pastel','gold','volcano','ocean','miami','toxic','chroma','prism'];
    35→    35→
    36→    36→/**
    37→    37→ * Écoute public-rewards et construit la map publicId → rewardType.
    38→    38→ * Le skin suit le PUBLIC ID (identité stable) plutôt que l'alias (changeant).
    39→    39→ * On lit data.publicId directement; en fallback on essaie data.username contre
    40→    40→ * le username du profil courant.
    41→    41→ */
    42→    42→function loadVipForProfile() {
    43→    43→  if (_vipUnsub) return; // déjà abonné
    44→    44→  try {
    45→    45→    _vipUnsub = onSnapshot(collection(db, "public-rewards"), (snap) => {
    46→    46→      vipPlayersByPid = new Map();
    47→    47→      // fallback: username → rewardType (pour les docs sans publicId direct)
    48→    48→      const usernameToType = new Map();
    49→    49→      snap.forEach((docSnap) => {
    50→    50→        const data = docSnap.data();
    51→    51→        const rewardType = data.activeType || data.type || null;
    52→    52→        if (!rewardType || data.activated === false) return;
    53→    53→        if (data.publicId) vipPlayersByPid.set(String(data.publicId), rewardType);
    54→    54→        if (data.username) usernameToType.set(data.username, rewardType);
    55→    55→      });
    56→    56→      // Re-applique le skin sur le hero si on a un profil
    57→    57→      // En mode visualisation publique, on utilise le profil virtuel du joueur consulté
    58→    58→      // (son publicId) plutôt que le profil propre de l'utilisateur courant.
    59→    59→      if (viewingPublicId) {
    60→    60→        applyProfileSkin({ username: viewingUsername, publicId: viewingPublicId }, usernameToType);
    61→    61→      } else if (currentProfile) {
    62→    62→        applyProfileSkin(currentProfile, usernameToType);
    63→    63→      }
    64→    64→    }, (err) => {
    65→    65→      console.warn("[profile] VIP listener error (non-critique):", err.message);
    66→    66→    });
    67→    67→  } catch (e) {
    68→    68→    console.warn("[profile] loadVipForProfile error:", e);
    69→    69→  }
    70→    70→}

    76→    76→function applyProfileSkin(profile, usernameToTypeFallback) {

    83→    83→  if (rewardType && NEW_SKIN_TYPES.includes(rewardType)) {
    84→    84→    nameEl.className = `rgb-${rewardType}`;
    85→    85→  } else if (rewardType) {
    86→    86→    nameEl.className = `player-${rewardType}`;
    87→    87→  } else {
    88→    88→    nameEl.className = "";
    89→    89→  }

   139→   139→/**
   140→   140→ * Détecte si l'URL demande de visualiser le profil PUBLIC d'un autre joueur.
   141→   141→ * Format: profile.html?player=NAME&publicId=XXXXXXXX
   142→   142→ * Si le publicId correspond à celui de l'utilisateur courant, on ignore
   143→   143→ * (c'est son propre profil — flux normal).
   144→   144→ */
   145→   145→function getPublicProfileRequest() {
   146→   146→  const params = new URLSearchParams(window.location.search);
   147→   147→  const pid = (params.get("publicId") || "").trim();
   148→   148→  const name = (params.get("player") || "").trim();
   149→   149→  if (pid && /^[A-Za-z0-9]{8}$/.test(pid)) {
   150→   150→    return { publicId: pid, username: name || pid };
   151→   151→  }
   152→   152→  return null;
   153→   153→}

   155→   155→onAuthStateChanged(auth, async (user) => {
   156→   156→  // ── Cas 1 : visualisation du profil public d'un autre joueur ──
   157→   157→  // On vérifie l'URL AVANT toute logique d'auth, car cela doit fonctionner
   158→   158→  // même si l'utilisateur n'est pas connecté.
   159→   159→  const pubReq = getPublicProfileRequest();
   160→   160→  if (pubReq) {
   161→   161→    // Lecture du propre profil de l'utilisateur courant (s'il est connecté)
   162→   162→    // pour détecter s'il visualise son PROPRE profil → flux normal.
   163→   163→    let ownProfile = null;
   164→   164→    if (user) {
   165→   165→      try {
   166→   166→        const snap = await getDoc(doc(db, "users", user.uid));
   167→   167→        if (snap.exists()) ownProfile = snap.data();
   168→   168→      } catch (e) {
   169→   169→        console.warn("[profile] Firestore read error (own, non-bloquant):", e.message);
   170→   170→      }
   171→   171→    }
   172→   172→
   173→   173→    if (ownProfile && ownProfile.publicId === pubReq.publicId) {
   174→   174→      // L'utilisateur visualise son propre profil → flux normal (on nettoie l'URL)
   175→   175→      history.replaceState(null, "", window.location.pathname);
   176→   176→      currentUser = user;
   177→   177→      currentProfile = ownProfile;
   178→   178→      updateSidebarUI(user, ownProfile);
   179→   179→      showView("profile-main");
   180→   180→      renderHero(user, ownProfile);
   181→   181→      loadVipForProfile();
   182→   182→      await loadStats(ownProfile.publicId);
   183→   183→      return;
   184→   184→    }
   185→   185→
   186→   186→    // ── Profil d'un AUTRE joueur (ou visiteur non connecté) ──
   187→   187→    currentUser = user; // peut être null
   188→   188→    currentProfile = ownProfile; // pour la sidebar (peut être null)
   189→   189→    updateSidebarUI(user, ownProfile);
   190→   190→    viewingPublicId = pubReq.publicId;
   191→   191→    viewingUsername = pubReq.username;
   192→   192→    showView("profile-main");
   193→   193→    renderPublicProfile(pubReq.username, pubReq.publicId);
   194→   194→    loadVipForProfile();
   195→   195→    await loadStats(pubReq.publicId);
   196→   196→    return;
   197→   197→  }
   198→   198→
   199→   199→  // ── Cas 2 : pas de ?publicId dans l'URL → flux normal ──
   200→   200→  if (!user) {
   201→   201→    currentUser = null;
   202→   202→    currentProfile = null;
   203→   203→    updateSidebarUI(null);
   204→   204→    showView("profile-gate");
   205→   205→    return;
   206→   206→  }
   207→   207→
   208→   208→  currentUser = user;
   209→   209→
   210→   210→  // Read Firestore profile
   211→   211→  let profile = null;
   212→   212→  try {
   213→   213→    const snap = await getDoc(doc(db, "users", user.uid));
   214→   214→    if (snap.exists()) profile = snap.data();
   215→   215→  } catch (e) {
   216→   216→    console.error("[profile] Firestore read error:", e);
   217→   217→    showToast("Erreur de lecture du profil (Firestore).", "error");
   218→   218→  }
   219→   219→
   220→   220→  currentProfile = profile;
   221→   221→  updateSidebarUI(user, profile);
   222→   222→
   223→   223→  if (!profile || !profile.publicId) {
   224→   224→    // New user → setup form
   225→   225→    showView("profile-setup");
   226→   226→    return;
   227→   227→  }
   228→   228→
   229→   229→  // Returning user with publicId → fetch & display stats
   230→   230→  showView("profile-main");
   231→   231→  renderHero(user, profile);
   232→   232→  // Lance l'écoute VIP (skin par publicId) — re-applique le skin dès que les rewards arrivent
   233→   233→  loadVipForProfile();
   234→   234→  await loadStats(profile.publicId);
   235→   235→});

   332→   332→async function loadStats(publicId) {
   333→   333→  // Reset stats list to loading
   334→   334→  setText("stat-week-rank", "This week rank: …");
   335→   335→  setText("stat-week-score", "This week score: …");
   336→   336→  setText("stat-alltime", "All-time score: …");
   337→   337→  const recentEl = document.getElementById("profile-recent-games");
   338→   338→  if (recentEl) recentEl.innerHTML = `<div class="pf-empty">Chargement…</div>`;
   339→   339→  hideError();
   340→   340→
   341→   341→  // Kick off ELO lookup (ranked.json) in parallel
   342→   342→  const eloPromise = getRankedEntry(publicId);
   343→   343→
   344→   344→  // Kick off recent games fetch in parallel (separate endpoint)
   345→   345→  // /public/player/{id} returns aggregated stats only (no games array).
   346→   346→  // /public/player/{id}/games returns the actual recent games list with
   347→   347→  // result (victory/defeat) already included — no need for per-game fetch.
   348→   348→  const recentGamesPromise = fetchRecentGames(publicId);
   349→   349→
   350→   350→  let playerData;
   351→   351→  try {
   352→   352→    playerData = await fetchOpenFront(`/public/player/${encodeURIComponent(publicId)}`);
   353→   353→  } catch (e) {
   354→   354→    console.error("[profile] OpenFront API error:", e);
   355→   355→    showError(`Impossible de charger les statistiques depuis l'API OpenFront.`);
   356→   356→    setText("stat-week-rank", "This week rank: —");
   357→   357→    setText("stat-week-score", "This week score: —");
   358→   358→    setText("stat-alltime", "All-time score: —");
   359→   359→    return;
   360→   360→  }
   361→   361→
   362→   362→  if (!playerData) {
   363→   363→    showError("Réponse vide de l'API OpenFront.");
   364→   364→    return;
   365→   365→  }
   366→   366→
   367→   367→  // NOTE: /public/player/{id} no longer returns a `games` array.
   368→   368→  // Recent games come from the separate /games endpoint (recentGamesPromise).
   369→   369→  const games = [];
   370→   370→  const stats = computeStats(games, playerData.stats || {});
   371→   371→
   372→   372→  // Compute week stats (games in last 7 days)
   373→   373→  const now = Date.now();
   374→   374→  const weekAgo = now - 7 * 24 * 60 * 60 * 1000;
   375→   375→  const weekGames = games.filter(g => {
   376→   376→    const t = new Date(g.start || 0).getTime();
   377→   377→    return !isNaN(t) && t >= weekAgo && t <= now;
   378→   378→  });
   379→   379→  const weekWins = weekGames.length; // approximate: count week games as "score"
   380→   380→  const weekRank = "—"; // not available without global calc
   381→   381→
   382→   382→  // All-time score: wins * 1 + games played (simple heuristic)
   383→   383→  const allTimeScore = stats.wins * 4 + (stats.total - stats.wins);
   384→   384→
   385→   385→  // Breakdown by mode (from stats tree)
   386→   386→  const breakdown = computeModeBreakdown(playerData.stats || {});
   387→   387→  const detail = [];
   388→   388→  if (breakdown.FFA) detail.push("FFA: " + breakdown.FFA);
   389→   389→  if (breakdown.Team) detail.push("Team: " + breakdown.Team);
   390→   390→  if (breakdown.Duos) detail.push("Duos: " + breakdown.Duos);
   391→   391→  if (breakdown.Trios) detail.push("Trios: " + breakdown.Trios);
   392→   392→  if (breakdown.Quads) detail.push("Quads: " + breakdown.Quads);
   393→   393→  const detailStr = detail.length ? " (" + detail.join(", ") + ")" : "";
   394→   394→
   395→   395→  setText("stat-week-rank", `This week rank: #${weekRank}`);
   396→   396→  setText("stat-week-score", `This week score: ${weekWins}`);
   397→   397→  setText("stat-alltime", `All-time score: ${allTimeScore} (${stats.wins} wins${detailStr})`);
   398→   398→
   399→   399→  // ELO from ranked.json (1v1)
   400→   400→  const ranked1v1 = await eloPromise;
   401→   401→  const eloLine = document.getElementById("stat-elo-line");
   402→   402→  if (eloLine) {
   403→   403→    if (ranked1v1 && ranked1v1.elo != null) {
   404→   404→      eloLine.textContent = `ELO 1v1: ${ranked1v1.elo} (Peak: ${ranked1v1.peakElo ?? '—'}) — Rank #${ranked1v1.rank}`;
   405→   405→      eloLine.style.display = "list-item";
   406→   406→    } else {
   407→   407→      eloLine.style.display = "none";
   408→   408→    }
   409→   409→  }
   410→   410→  
   411→   411→  // ELO 2v2 from ranked.json
   412→   412→  const ranked2v2 = await getRankedEntry(publicId, "2v2");
   413→   413→  const elo2v2Line = document.getElementById("stat-elo-2v2-line");
   414→   414→  if (elo2v2Line) {
   415→   415→    if (ranked2v2 && ranked2v2.elo != null) {
   416→   416→      elo2v2Line.textContent = `ELO 2v2: ${ranked2v2.elo} (Peak: ${ranked2v2.peakElo ?? '—'}) — Rank #${ranked2v2.rank}`;
   417→   417→      elo2v2Line.style.display = "list-item";
   418→   418→    } else {
   419→   419→      elo2v2Line.style.display = "none";
   420→   420→    }
   421→   421→  }

   460→   460→function computeModeBreakdown(statsTree) {
   461→   461→  const out = { FFA: 0, Team: 0, Duos: 0, Trios: 0, Quads: 0 };
   462→   462→  if (!statsTree || typeof statsTree !== "object") return out;
   463→   463→  for (const catKey of Object.keys(statsTree)) {
   464→   464→    const cat = statsTree[catKey];
   465→   465→    if (!cat || typeof cat !== "object") continue;
   466→   466→    for (const modeKey of Object.keys(cat)) {
   467→   467→      const mode = cat[modeKey];
   468→   468→      if (!mode || typeof mode !== "object") continue;
   469→   469→      let wins = 0;
   470→   470→      for (const diffKey of Object.keys(mode)) {
   471→   471→        const diff = mode[diffKey];
   472→   472→        if (diff && typeof diff === "object" && diff.wins != null) {
   473→   473→          wins += parseInt(diff.wins, 10) || 0;
   474→   474→        }
   475→   475→      }
   476→   476→      if (modeKey === "Free For All") out.FFA += wins;
   477→   477→      else if (modeKey === "Team") {
   478→   478→        // Try to break down by playerTeams if available
   479→   479→        out.Team += wins;
   480→   480→      }
   481→   481→    }
   482→   482→  }
   483→   483→  return out;
   484→   484→}

   486→   486→function computeStats(games, statsTree) {
   487→   487→  // Wins: sum all "wins" fields across the stats tree (Private/Public/Ranked → mode → difficulty)
   488→   488→  let wins = 0;
   489→   489→  let total = 0;
   490→   490→  if (statsTree && typeof statsTree === "object") {
   491→   491→    for (const catKey of Object.keys(statsTree)) {
   492→   492→      const cat = statsTree[catKey];
   493→   493→      if (!cat || typeof cat !== "object") continue;
   494→   494→      for (const modeKey of Object.keys(cat)) {
   495→   495→        const mode = cat[modeKey];
   496→   496→        if (!mode || typeof mode !== "object") continue;
   497→   497→        for (const diffKey of Object.keys(mode)) {
   498→   498→          const diff = mode[diffKey];
   499→   499→          if (!diff || typeof diff !== "object") continue;
   500→   500→          if (diff.wins != null) wins += parseInt(diff.wins, 10) || 0;
   501→   501→          if (diff.total != null) total += parseInt(diff.total, 10) || 0;
   502→   502→          else if (diff.wins != null && diff.losses != null) {
   503→   503→            total += (parseInt(diff.wins, 10) || 0) + (parseInt(diff.losses, 10) || 0);
   504→   504→          }
   505→   505→        }
   506→   506→      }
   507→   507→    }
   508→   508→  }
   509→   509→
   510→   510→  // Fallback: if stats tree has no totals, use games.length
   511→   511→  if (total === 0 && games.length > 0) total = games.length;
   512→   512→
   513→   513→  // Unique maps + favourite map
   514→   514→  const mapCounts = {};
   515→   515→  let lastGame = null;
   516→   516→  for (const g of games) {
   517→   517→    if (g.map) mapCounts[g.map] = (mapCounts[g.map] || 0) + 1;
   518→   518→    if (g.start) {
   519→   519→      const d = new Date(g.start).getTime();
   520→   520→      if (!isNaN(d) && (lastGame === null || d > lastGame)) lastGame = d;
   521→   521→    }
   522→   522→  }
   523→   523→  const uniqueMaps = Object.keys(mapCounts).length;
   524→   524→  let favMap = null;
   525→   525→  let favCount = 0;
   526→   526→  for (const [m, c] of Object.entries(mapCounts)) {
   527→   527→    if (c > favCount) { favMap = m; favCount = c; }
   528→   528→  }
   529→   529→  const lastGameIso = lastGame ? new Date(lastGame).toISOString() : null;
   530→   530→
   531→   531→  return { wins, total, uniqueMaps, favMap, lastGame: lastGameIso };
   532→   532→}

   613→   613→/** Format duration in seconds as M:SS or H:MM:SS */
   614→   614→function formatDuration(seconds) {
   615→   615→  const s = Math.floor(Number(seconds) || 0);
   616→   616→  if (s < 60) return `${s}s`;
   617→   617→  const m = Math.floor(s / 60);
   618→   618→  const rs = s % 60;
   619→   619→  if (m < 60) return `${m}:${String(rs).padStart(2, "0")}`;
   620→   620→  const h = Math.floor(m / 60);
   621→   621→  const rm = m % 60;
   622→   622→  return `${h}:${String(rm).padStart(2, "0")}:${String(rs).padStart(2, "0")}`;
   623→   623→}
