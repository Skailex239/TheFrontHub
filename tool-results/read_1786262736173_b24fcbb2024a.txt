   100→function loadRuns() {
   101→  try {
   102→    // 1. Try full file first (fastest, raw array — local only, never committed)
   103→    if (fs.existsSync(RUNS_FULL_FILE)) {
   104→      const raw = JSON.parse(fs.readFileSync(RUNS_FULL_FILE, "utf8"));
   105→      return Array.isArray(raw) ? raw : (raw.runs || []);
   106→    }
   107→    // 2. Try uncompressed JSON (gitignored, may exist locally)
   108→    if (fs.existsSync(RUNS_FILE)) {
   109→      const raw = JSON.parse(fs.readFileSync(RUNS_FILE, "utf8"));
   110→      return Array.isArray(raw) ? raw : (raw.runs || []);
   111→    }
   112→    // 3. Try gzipped file (THIS is the one that's actually committed to the repo!)
   113→    const gzPath = RUNS_FILE + ".gz";
   114→    if (fs.existsSync(gzPath)) {
   115→      const gzipped = fs.readFileSync(gzPath);
   116→      const decompressed = zlib.gunzipSync(gzipped);
   117→      const raw = JSON.parse(decompressed.toString("utf8"));
   118→      return Array.isArray(raw) ? raw : (raw.runs || []);
   119→    }
   120→  } catch (e) {
   121→    console.warn(`[sync] ⚠️ loadRuns error: ${e.message}`);
   122→  }
   123→  return [];
   124→}

   126→function saveRuns(runs) {
   127→  const meta = {
   128→    totalCount: runs.length,
   129→    lastUpdate: new Date().toISOString(),
   130→    latestCommit: currentLatestCommit,
   131→  };
   132→
   133→  // Fichier public (sans URL, plus léger)
   134→  const cleanedRuns = runs.map(({ url, ...rest }) => rest);
   135→  const publicOutput = { ...meta, runs: cleanedRuns };
   136→  const jsonString = JSON.stringify(publicOutput);
   137→
   138→  fs.writeFileSync(RUNS_FILE, jsonString);
   139→  const gzipped = zlib.gzipSync(jsonString);
   140→  fs.writeFileSync(RUNS_FILE + ".gz", gzipped);
   141→
   142→  // Backup complet (toutes les runs + URLs), mis à jour à chaque sync
   143→  const backupOutput = { ...meta, runs };
   144→  const backupString = JSON.stringify(backupOutput);
   145→  fs.writeFileSync(RUNS_BACKUP_FILE, backupString);
   146→  try {
   147→    fs.writeFileSync(RUNS_BACKUP_FILE + ".gz", zlib.gzipSync(backupString));
   148→  } catch (e) {
   149→    console.warn("[sync] ⚠️ Impossible d'écrire runs_backup.json.gz:", e.message);
   150→  }
   151→
   152→  // Archive interne (tableau brut, utilisé par loadRuns)
   153→  try {
   154→    fs.writeFileSync(RUNS_FULL_FILE, JSON.stringify(runs));
   155→  } catch (e) {
   156→    console.warn("[sync] ⚠️ Impossible d'écrire runs_full.json:", e.message);
   157→  }
   158→
   159→  // ── Optimized public payload (tiny instead of 15+ MB) ──
   160→  // Only includes: top 25 per map + 50 most recent runs
   161→  // Uses compact array format to minimize size
   162→  try {
   163→    const byMap = {};
   164→    const mapTotals = {}; // NEW: Store true count of runs per map
   165→    runs.forEach(r => { 
   166→      if (!byMap[r.map]) byMap[r.map] = []; 
   167→      byMap[r.map].push(r); 
   168→      mapTotals[r.map] = (mapTotals[r.map] || 0) + 1;
   169→    });
   170→    const topPerMap = [];
   171→    Object.entries(byMap).forEach(([map, mapRuns]) => {
   172→      mapRuns.sort((a, b) => a.duration_s - b.duration_s);
   173→      topPerMap.push(...mapRuns.slice(0, 25));
   174→    });
   175→
   176→    const allSorted = [...runs].sort((a, b) => new Date(b.timestamp) - new Date(a.timestamp));
   177→    const recent50 = allSorted.slice(0, 50);
   178→
   179→    const seen = new Set();
   180→    const merged = [];
   181→    [...recent50, ...topPerMap].forEach(r => {
   182→      if (!seen.has(r.id)) { seen.add(r.id); merged.push(r); }
   183→    });
   184→
   185→    // Compact format: array of arrays instead of array of objects
   186→    // Key map: [id, player, playerId, map, duration_s, difficulty, bots, players, timestamp]
   187→    const publicPayload = {
   188→      t: meta.totalCount,
   189→      u: meta.lastUpdate,
   190→      c: meta.latestCommit,
   191→      m: mapTotals, // NEW: Include map totals in the payload
   192→      k: ['id', 'player', 'playerId', 'map', 'duration_s', 'difficulty', 'bots', 'players', 'timestamp'],
   193→      r: merged.map(r => [r.id, r.player, r.playerId || '', r.map, r.duration_s, r.difficulty || '', r.bots || 0, r.players || 0, r.timestamp])
   194→    };
   195→
   196→    const publicJson = JSON.stringify(publicPayload);
   197→    const publicFile = RUNS_FILE.replace('.json', '_public.json');
   198→    fs.writeFileSync(publicFile, publicJson);
   199→    fs.writeFileSync(publicFile + '.gz', zlib.gzipSync(publicJson));
   200→    console.log(`[sync] 🚀 Public payload: ${(publicJson.length / 1024).toFixed(0)} KB raw, ${(zlib.gzipSync(publicJson).length / 1024).toFixed(0)} KB gzipped (${merged.length} runs, top 25/map + 50 recent)`);
   201→  } catch (e) {
   202→    console.warn("[sync] ⚠️ Impossible de générer le payload public:", e.message);
   203→  }
   204→
   205→  console.log(
   206→    `[sync] 💾 ${runs.length} runs — public ${(jsonString.length / 1024 / 1024).toFixed(2)} Mo, ` +
   207→    `backup ${(backupString.length / 1024 / 1024).toFixed(2)} Mo`
   208→  );
   209→}

   235→// ── Fetch avec retry et gestion 429 ──────────────────────────────────────────
   236→async function fetchWithRetry(url, retries = 3) {
   237→  for (let attempt = 0; attempt <= retries; attempt++) {
   238→    const controller = new AbortController();
   239→    const timer = setTimeout(() => controller.abort(), FETCH_TIMEOUT);
   240→    try {
   241→      const res = await openFrontFetch(url, { signal: controller.signal });
   242→      clearTimeout(timer);
   243→      if (res.status === 429) {
   244→        const wait = DELAY_429 * (attempt + 1);
   245→        console.log(`[rate-limit] 429 — attente ${wait}ms (tentative ${attempt + 1}/${retries})`);
   246→        await sleep(wait);
   247→        continue;
   248→      }
   249→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   250→      return await res.json();
   251→    } catch (e) {
   252→      clearTimeout(timer);
   253→      if (e.name === "AbortError") {
   254→        if (attempt < retries) { await sleep(500); continue; }
   255→        throw new Error("Timeout");
   256→      }
   257→      if (attempt === retries) throw e;
   258→      await sleep(500 * (attempt + 1));
   259→    }
   260→  }
   261→}

   277→/** Découpe [rangeStart, rangeEnd] en intervalles de 30 secondes */
   278→function buildWindows30s(rangeStart, rangeEnd) {
   279→  const windows = [];
   280→  for (let end = rangeEnd.getTime(); end > rangeStart.getTime(); end -= WINDOW_MS) {

   303→async function processOneGame(game, seen, runs, runIds) {
   304→  const gameId = game.game;
   305→  try {
   306→    const raw = await fetchGameDetail(gameId);
   307→    seen.add(gameId);

   310→      runs.push(run);
   311→      runIds.add(run.id);
   312→      const mins = Math.floor(run.duration_s / 60);
   313→      const secs = String(run.duration_s % 60).padStart(2, "0");
   314→      const modeLabel = syncMode === "compact" ? "compact" : "sync";
   315→      console.log(`[${modeLabel}] ✅ ${run.player} — ${run.map} — ${mins}m${secs}s (${run.difficulty}, ${run.players}p)`);
   316→      return 1;
   317→    }
   318→  } catch (e) {
   319→    return { error: e, gameId };
   320→  }
   321→  return 0;
   322→}

   324→// ── Traitement d'un lot de parties (parallèle si exemption) ───────────────────
   325→async function processGames(games, seen, runs, runIds) {
   326→  const unseen = games.filter(g => g.game && !seen.has(g.game));
   327→  if (unseen.length === 0) return 0;
   328→
   329→  let newRuns = 0;
   330→  let errors = 0;
   331→
   332→  console.log(
   333→    `[sync] ${unseen.length} parties à détailler (×${DETAIL_CONCURRENCY} parallèle${HAS_EXEMPTION ? ", mode rapide" : ""})`
   334→  );
   335→
   336→  for (let i = 0; i < unseen.length; i += DETAIL_CONCURRENCY) {
   337→    const chunk = unseen.slice(i, i + DETAIL_CONCURRENCY);
   338→    const results = await Promise.all(
   339→      chunk.map(game => processOneGame(game, seen, runs, runIds))
   340→    );
   341→    for (const r of results) {
   342→      if (typeof r === "number") newRuns += r;
   343→      else if (r?.error) {
   344→        errors++;
   345→        if (errors <= 5) console.warn(`[sync] ⚠️ ${r.gameId}: ${r.error.message}`);
   346→      }
   347→    }
   348→    if (BATCH_DELAY_NORMAL > 0) await sleep(BATCH_DELAY_NORMAL);
   349→  }
   350→
   351→  if (errors > 5) console.log(`[sync] ... et ${errors - 5} autres erreurs`);
   352→  return newRuns;
   353→}

   360→async function syncRecent() {
   361→  console.log(`[sync] 🔄 Sync récente — ${new Date().toISOString()}`);
   362→  const seen = loadSeen();
   363→  const runs = loadRuns();
   364→  const runIds = new Set(runs.map(r => r.id));

   366→
   367→  const now = new Date();
   368→  const cp = loadCheckpoints();
   369→  const lastSync = cp.last_sync_time ? parseInt(cp.last_sync_time, 10) : 0;
   370→  const agoMs = Math.max(now.getTime() - RECENT_MAX_MS, lastSync - RECENT_OVERLAP_MS);
   371→  const ago = new Date(agoMs);
   372→  const windowMin = Math.round((now - ago) / 60_000);
   373→
   374→  const windows = buildWindows30s(ago, now);
   375→  console.log(
   376→    `[sync] ${windows.length} fenêtres de 30s (~${windowMin} min, max 3h, filtre Public FFA ≥10p)`
   377→  );
   378→
   379→  for (const { start, end } of windows) {
   380→    const games = await fetchGamesInWindow(start, end);
   381→    if (games.length > 0) {
   382→      if (games.length >= WINDOW_SATURATION_THRESHOLD) {
   383→        console.log(
   384→          `[sync] ⚠️ Fenêtre saturée (${games.length}) ${start.toISOString().slice(11, 19)} — possible troncature`
   385→        );
   386→      }
   387→      totalNew += await processGames(games, seen, runs, runIds);
   388→    }
   389→    if (WINDOW_DELAY > 0) await sleep(WINDOW_DELAY);
   390→  }
   391→
   392→  if (totalNew > 0) saveRuns(runs);
   393→  saveSeen(seen);
   394→
   395→  cp.last_sync_time = String(Date.now());
   396→  saveCheckpoints(cp);
   397→
   398→  logApiStats("sync-recent");
   399→  console.log(`[sync] ✅ Sync récente terminée — ${totalNew} nouveaux runs`);
   400→  return totalNew;
   401→}

   476→
   477→  console.log(`[history] 🕐 Reprise depuis ${new Date(resumeFrom).toISOString()}`);
   478→
   479→  const rangeEnd = new Date(resumeFrom);
   480→  const rangeStart = new Date(oldest);
   481→  const windows = buildWindows30s(rangeStart, rangeEnd);
   482→
   483→  const toProcess = Math.min(windows.length, maxWindows);
   484→  console.log(`[history] ${windows.length.toLocaleString()} fenêtres restantes — traitement de ${toProcess} (max ${maxWindows})`);
   485→
   486→  const seen = loadSeen();
   487→  const runs = loadRuns();
   488→  const runIds = new Set(runs.map(r => r.id));
   489→  let totalRuns = 0;
   490→  let oldestReached = resumeFrom;
   491→  let saturatedWindows = 0;
   492→
   493→  for (let i = 0; i < toProcess; i++) {
   494→    const { start, end } = windows[i];
   495→    try {
   496→      const games = await fetchGamesInWindow(start, end);
   497→      if (games.length > 0) {
   498→        if (games.length >= WINDOW_SATURATION_THRESHOLD) {
   499→          saturatedWindows++;
   500→          if (saturatedWindows <= 3) {
   501→            console.log(
   502→              `[history] ⚠️ Fenêtre saturée (${games.length}): ${start.toISOString().slice(0, 16)}`
   503→            );
   504→          }
   505→        }
   506→        const added = await processGames(games, seen, runs, runIds);
   507→        totalRuns += added;
   508→        if (added > 0) {
   509→          console.log(`[history] +${added} runs (${start.toISOString().slice(0, 10)} ${start.toISOString().slice(11, 16)})`);
   510→        }
   511→      }
   512→      oldestReached = end.getTime();
   513→    } catch (e) {
   514→      console.warn(`[history] ⚠️ Erreur fenêtre ${start.toISOString()}: ${e.message}`);
   515→    }
   516→
   517→    if (WINDOW_DELAY > 0) await sleep(WINDOW_DELAY);
   518→
   519→    // Checkpoint périodique
   520→    if ((i + 1) % CHECKPOINT_EVERY === 0 || i === toProcess - 1) {
   521→      cp.history_oldest_reached = String(oldestReached);
   522→      cp.history_saturated_windows = (cp.history_saturated_windows || 0) + saturatedWindows;
   523→      saveCheckpoints(cp);
   524→      saveSeen(seen);
   525→      if (totalRuns > 0) saveRuns(runs);
   526→
   527→      const pct = Math.round(((now - oldestReached) / (now - oldest)) * 100);
   528→      console.log(`[history] 💾 ${i + 1}/${toProcess} fenêtres — ${totalRuns} runs — ${pct}% de l'historique`);
   529→    }
   530→  }
   531→
   532→  // Sauvegarde finale
   533→  if (totalRuns > 0) saveRuns(runs);
   534→  saveSeen(seen);
   535→
   536→  if (saturatedWindows > 0) {
   537→    console.log(`[history] ⚠️ ${saturatedWindows} fenêtres saturées détectées — certains runs ont peut-être été manqués`);
   538→  }
   539→
   540→  if (windows.length > maxWindows) {
   541→    console.log(`[history] ⏹️ Limite atteinte — reprendra au prochain run (reste: ${(windows.length - toProcess).toLocaleString()} fenêtres)`);
   542→  } else {
   543→    console.log(`[history] ✅ Historique terminé — ${totalRuns} runs insérés`);
   544→  }
   545→
   546→  return totalRuns;
   547→}

   594→async function enrichPlayerIds() {
   595→  const runs = loadRuns();
   596→  const missing = runs.filter(r => !r.playerId);
   597→  console.log(`[enrich] ${missing.length} runs sans playerId sur ${runs.length} total`);

   600→    console.log("[enrich] ✅ Tous les runs ont déjà un playerId");
   601→    return;
   602→  }
   603→
   604→  let fixed = 0, errors = 0;
   605→  const ENRICH_BATCH = 250; // Nombre de runs traités par exécution
   606→  
   607→  const toProcess = missing.slice(0, ENRICH_BATCH);
   608→  console.log(`[enrich] Traitement de ${toProcess.length} runs (max ${ENRICH_BATCH} par exécution)...`);
   609→
   610→  for (const run of toProcess) {
   611→    try {
   612→      const raw = await fetchWithRetry(`${API_BASE}/public/game/${run.id}?turns=false`);
   613→      const info = raw.info || raw;
   614→      const players = info.players || [];
   615→      const winner = info.winner;
   616→      if (winner && Array.isArray(winner) && winner.length >= 2) {
   617→        const wp = players.find(p => p.clientID === winner[1]);
   618→        if (wp && wp.clientID) {
   619→          run.playerId = wp.clientID;
   620→          fixed++;
   621→        }
   622→      }
   623→    } catch (e) {
   624→      errors++;
   625→    }
   626→    await sleep(150);
   627→  }
   628→
   629→  console.log(`[enrich] ✅ ${fixed} runs enrichis, ${errors} erreurs`);
   630→  if (missing.length > ENRICH_BATCH) {
   631→    console.log(`[enrich] ⏳ Reste ${missing.length - ENRICH_BATCH} runs — relancer 'node sync.js enrich'`);
   632→  }
   633→
   634→  saveRuns(runs);
   635→}
