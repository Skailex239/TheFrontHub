     1→/**
     2→ * dashboard.js — Contrôleur du Tableau de bord TheFrontHub.
     3→ *
     4→ * Architecture (NO SYNC — le site fait les requêtes API lui-même) :
     5→ *
     6→ *   1. ranked.json (auto-synced par GitHub Actions) → ranked wins carrière
     7→ *      pour le top 100 1v1 + top 100 2v2. Servé depuis GitHub Pages.
     8→ *
     9→ *   2. Firebase public-aliases (Firestore REST API) → liste des joueurs
    10→ *      connectés via Google/Discord qui ont lié leur Public ID.
    11→ *
    12→ *   3. OpenFront API (via proxy Cloudflare Worker ou /api/openfront en dev)
    13→ *      → pour chaque joueur connecté, on pagine TOUTES ses parties et on
    14→ *      compte les victoires par catégorie (FFA casual, FFA classé, Team
    15→ *      casual, Team classé). Le proxy ajoute le header x-skailex-access
    16→ *      côté serveur (exemption de rate-limit).
    17→ *
    18→ *   Merge :
    19→ *     - Joueurs ranked-only : ranked wins de ranked.json, casual = 0
    20→ *     - Joueurs connectés : ranked wins = max(ranked.json, API carrière),
    21→ *       casual wins = API live
    22→ *
    23→ *   Cache : les stats live sont mises en cache dans localStorage (30 min)
    24→ *   pour éviter de re-fetcher à chaque visite.
    25→ *
    26→ * Barème :
    27→ *   FFA casual  = +10  ·  FFA classé (1v1) = +1
    28→ *   Team casual = +5   ·  Team classé (2v2) = +1
    29→ *   (ranked = 1 pt, PAS en plus du FFA/Team)
    30→ *
    31→ * Auth : importe auth.js (Firebase) et écoute onAuthStateChanged pour brancher
    32→ * la sidebar (login-btn / user-badge). Définit sur window les handlers utilisés
    33→ * par les onclick du HTML : toggleAuthModal, handleLogin, handleLogout,
    34→ * toggleUserDropdown, goToProfilePage, closeProfileModal,
    35→ * startOwnershipVerification, confirmOwnershipVerification,
    36→ * cancelOwnershipVerification.
    37→ */

    87→function escapeHtml(s) {
    88→  if (s == null) return "";
    89→  return String(s)
    90→    .replace(/&/g, "&amp;")
    91→    .replace(/</g, "&lt;")
    92→    .replace(/>/g, "&gt;")
    93→    .replace(/"/g, "&quot;")
    94→    .replace(/'/g, "&#39;");
    95→}

   101→function initials(name) {
   102→  if (!name) return "?";
   103→  const clean = name.replace(/^\[[^\]]+\]\s*/, "").trim();
   104→  const parts = clean.split(/[\s_-]+/).filter(Boolean);
   105→  const letters = parts.length >= 2 ? parts[0][0] + parts[1][0] : clean.slice(0, 2);
   106→  return letters.toUpperCase();
   107→}

   149→/** Charge la liste des joueurs connectés depuis Firebase public-aliases. */
   150→async function loadConnectedPlayers() {
   151→  try {
   152→    const url = `${FIRESTORE_BASE}/public-aliases`;
   153→    const res = await fetch(url, { cache: "no-store" });
   154→    if (!res.ok) {
   155→      console.warn(`[dashboard] Firebase public-aliases: HTTP ${res.status}`);
   156→      return;
   157→    }
   158→    const data = await res.json();
   159→    const docs = data.documents || [];
   160→    const seen = new Set();
   161→    _connectedPlayers = docs.map((doc) => {
   162→      const fields = doc.fields || {};
   163→      const val = (f) => (f?.stringValue || f?.integerValue || "");
   164→      const publicId = val(fields.publicId);
   165→      return {
   166→        publicId,
   167→        username: val(fields.username) || publicId || "?",
   168→      };
   169→    })
   170→    // Filtrer : publicId valide = exactement 8 caractères alphanumériques
   171→    .filter((p) => /^[A-Za-z0-9]{8}$/.test(p.publicId))
   172→    // Dédoublonner par publicId (garder le premier)
   173→    .filter((p) => {
   174→      if (seen.has(p.publicId)) return false;
   175→      seen.add(p.publicId);
   176→      return true;
   177→    });
   178→    console.log(`[dashboard] Firebase: ${_connectedPlayers.length} joueurs connectés (sur ${docs.length} documents)`);
   179→  } catch (e) {
   180→    console.warn("[dashboard] Firebase indisponible:", e.message);
   181→  }
   182→}

   184→/**
   185→ * Pagine TOUTES les parties d'un joueur via le proxy OpenFront.
   186→ * Retourne [{ gameId, start, mode, rankedType, result, map, ... }].
   187→ */
   188→async function fetchAllPlayerGames(publicId) {
   189→  const allGames = [];
   190→  let cursor = null;
   191→  for (let page = 0; page < 500; page++) {
   192→    let apiPath = `/public/player/${encodeURIComponent(publicId)}/games`;
   193→    if (cursor) apiPath += `?cursor=${encodeURIComponent(cursor)}`;
   194→    let data;
   195→    try {
   196→      data = await fetchOpenFront(apiPath);
   197→    } catch (e) {
   198→      console.warn(`[dashboard] fetch games ${publicId} page ${page}:`, e.message);
   199→      break;
   200→    }
   201→    const results = data?.results || [];
   202→    if (results.length === 0) break;
   203→    allGames.push(...results);
   204→    if (!data.nextCursor) break;
   205→    cursor = data.nextCursor;
   206→    if (allGames.length >= MAX_GAMES_PER_PLAYER) break;
   207→  }
   208→  return allGames;
   209→}

   211→/** Classifie une game en catégorie. */
   212→function classifyGame(g) {
   213→  const mode = String(g.mode || "").toLowerCase();
   214→  const rt = String(g.rankedType || "").toLowerCase();
   215→  const isTeam =
   216→    mode === "team" ||
   217→    mode.startsWith("2v2") ||
   218→    mode.startsWith("3v3") ||
   219→    mode.startsWith("4v4") ||
   220→    rt === "2v2";
   221→  const isRanked = rt === "1v1" || rt === "2v2" || rt === "ranked";
   222→  if (isTeam) return isRanked ? "teamRanked" : "teamCasual";
   223→  return isRanked ? "ffaRanked" : "ffaCasual";
   224→}

   226→/** Calcule les wins globales + hebdo depuis une liste de games. */
   227→function computeWinsFromGames(games) {
   228→  const global = { ffaCasual: 0, ffaRanked: 0, teamCasual: 0, teamRanked: 0 };
   229→  const weekly = { ffaCasual: 0, ffaRanked: 0, teamCasual: 0, teamRanked: 0 };
   230→  const now = Date.now();
   231→  for (const g of games) {
   232→    if (g.result !== "victory") continue;
   233→    const cat = classifyGame(g);
   234→    global[cat]++;
   235→    if (g.start && now - new Date(g.start).getTime() < WEEKLY_MS) {
   236→      weekly[cat]++;
   237→    }
   238→  }
   239→  return { global, weekly };
   240→}

   242→/** Charge les stats live pour tous les joueurs connectés (avec cache, en parallèle). */
   243→async function loadLiveStats() {
   244→  if (_connectedPlayers.length === 0) {
   245→    _liveFetchDone = true;
   246→    return;
   247→  }
   248→
   249→  // ── 1. Charger le cache localStorage ──
   250→  let cached = {};
   251→  try {
   252→    const raw = localStorage.getItem(LIVE_CACHE_KEY);
   253→    if (raw) {
   254→      const parsed = JSON.parse(raw);
   255→      if (parsed && typeof parsed === "object") cached = parsed;
   256→    }
   257→  } catch (e) { /* ignore */ }
   258→
   259→  // ── 2. Séparer les joueurs en cache (frais) vs à fetcher ──
   260→  const toFetch = [];
   261→  for (const player of _connectedPlayers) {
   262→    const cacheEntry = cached[player.publicId];
   263→    const isFresh = cacheEntry && (Date.now() - cacheEntry.fetchedAt) < LIVE_CACHE_TTL;
   264→    if (isFresh) {
   265→      _liveStats[player.publicId] = cacheEntry;
   266→      _liveFetchProgress++;
   267→    } else {
   268→      toFetch.push(player);
   269→    }
   270→  }
   271→  // Premier rendu avec les données en cache
   272→  if (_liveFetchProgress > 0) mergeAndRender();
   273→
   274→  // ── 3. Fetch tous les joueurs en parallèle (exemption = 8 concurrent) ──
   275→  const fetchOne = async (player) => {
   276→    try {
   277→      const games = await fetchAllPlayerGames(player.publicId);
   278→      const wins = computeWinsFromGames(games);
   279→      const entry = {
   280→        publicId: player.publicId,
   281→        username: player.username,
   282→        games: games.slice(-20), // garder les 20 plus récentes pour le profil
   283→        global: wins.global,
   284→        weekly: wins.weekly,
   285→        totalGames: games.length,
   286→        fetchedAt: Date.now(),
   287→      };
   288→      _liveStats[player.publicId] = entry;
   289→      cached[player.publicId] = entry;
   290→      localStorage.setItem(LIVE_CACHE_KEY, JSON.stringify(cached));
   291→      console.log(`[dashboard] live: ${player.username} (${player.publicId}) → ${games.length} games, global=${JSON.stringify(wins.global)}`);
   292→    } catch (e) {
   293→      console.warn(`[dashboard] live fetch failed for ${player.publicId}:`, e.message);
   294→    }
   295→    _liveFetchProgress++;
   296→    mergeAndRender(); // rendu progressif après chaque joueur
   297→  };
   298→
   299→  // Lancer tous les fetchs en parallèle
   300→  await Promise.all(toFetch.map(fetchOne));
   301→  _liveFetchDone = true;
   302→}

   304→function updateLastUpdateLabel(ts) {
   305→  if (!ts || !lastUpdateEl) return;
   306→  const d = new Date(typeof ts === "number" ? ts : ts);
   307→  if (Number.isNaN(d.getTime())) return;
   308→  lastUpdateEl.textContent = "Mis à jour le " + new Intl.DateTimeFormat("fr-FR", {
   309→    day: "2-digit", month: "2-digit", year: "numeric",
   310→    hour: "2-digit", minute: "2-digit",
   311→  }).format(d);
   312→}

   318→function pointsFor(p) {
   319→  return (p.ffaCasualWins || 0) * PTS_FFA_CASUAL
   320→       + (p.ffaRankedWins || 0) * PTS_FFA_RANKED
   321→       + (p.teamCasualWins || 0) * PTS_TEAM_CASUAL
   322→       + (p.teamRankedWins || 0) * PTS_TEAM_RANKED;
   323→}

   325→/** Fusionne ranked.json (career ranked wins) + live API (casual wins). */
   326→function buildMergedPlayers() {
   327→  const byPid = new Map();
   328→  const isWeekly = _dashMode === "weekly";
   329→
   330→  // 1. ranked.json → ranked wins carrière (GLOBAL uniquement)
   331→  //    En weekly, on ne peut pas déduire les wins hebdo depuis ranked.json (career total only)
   332→  //    → les joueurs ranked-only ont 0 pts en weekly
   333→  if (_rankedData) {
   334→    const getOrCreate = (pid, name) => {
   335→      let e = byPid.get(pid);
   336→      if (!e) {
   337→        e = {
   338→          publicId: pid, username: name || pid, clan: null,
   339→          ffaCasualWins: 0, ffaRankedWins: 0,
   340→          teamCasualWins: 0, teamRankedWins: 0,
   341→          _hasLive: false,
   342→        };
   343→        byPid.set(pid, e);
   344→      }
   345→      return e;
   346→    };
   347→    for (const p of _rankedData["1v1"] || []) {
   348→      const nm = p.username || p.accountUsername || p.public_id;
   349→      const e = getOrCreate(p.public_id, nm);
   350→      // En weekly, on garde les career wins uniquement si le joueur n'a PAS de stats live
   351→      // (les stats live remplaceront par les vraies valeurs hebdo)
   352→      if (!isWeekly) e.ffaRankedWins = p.wins || 0;
   353→      if (nm && nm !== p.public_id) e.username = nm;
   354→    }
   355→    for (const p of _rankedData["2v2"] || []) {
   356→      const nm = p.username || p.accountUsername || p.public_id;
   357→      const e = getOrCreate(p.public_id, nm);
   358→      if (!isWeekly) e.teamRankedWins = p.wins || 0;
   359→      if (nm && nm !== p.public_id) e.username = nm;
   360→    }
   361→  }
   362→
   363→  // 2. Stats live → casual wins + ranked wins
   364→  for (const [pid, live] of Object.entries(_liveStats)) {
   365→    let e = byPid.get(pid);
   366→    if (!e) {
   367→      e = {
   368→        publicId: pid, username: live.username || pid, clan: null,
   369→        ffaCasualWins: 0, ffaRankedWins: 0,
   370→        teamCasualWins: 0, teamRankedWins: 0,
   371→        _hasLive: false,
   372→      };
   373→      byPid.set(pid, e);
   374→    }
   375→    const g = isWeekly ? live.weekly : live.global;
   376→    e.ffaCasualWins = g.ffaCasual;
   377→    e.teamCasualWins = g.teamCasual;
   378→    // Ranked wins :
   379→    //   - Global : max(ranked.json career, API live) — ranked.json est plus complet
   380→    //   - Weekly : API live uniquement (ranked.json n'a pas de breakdown hebdo)
   381→    if (isWeekly) {
   382→      e.ffaRankedWins = g.ffaRanked;
   383→      e.teamRankedWins = g.teamRanked;
   384→    } else {
   385→      e.ffaRankedWins = Math.max(e.ffaRankedWins || 0, g.ffaRanked);
   386→      e.teamRankedWins = Math.max(e.teamRankedWins || 0, g.teamRanked);
   387→    }
   388→    e._hasLive = true;
   389→    if (live.username && live.username !== pid) e.username = live.username;
   390→  }
   391→
   392→  // 3. Calcul des points + tri
   393→  const players = [...byPid.values()].map((p) => ({
   394→    ...p,
   395→    points: pointsFor(p),
   396→  }));
   397→  players.sort((a, b) => b.points - a.points);
   398→  return players;
   399→}

   410→function render() {
   411→  if (!_rankedData && _mergedPlayers.length === 0) {
   412→    view.innerHTML = `
   413→      <div class="dash-empty-state">
   414→        <div class="dash-empty-icon"><i data-icon="chart"></i></div>
   415→        <h3>Chargement…</h3>
   416→        <p>Récupération du classement…</p>
   417→      </div>`;
   418→    if (window.hydrateIcons) window.hydrateIcons(view);
   419→    return;
   420→  }
   421→
   422→  const isWeekly = _dashMode === "weekly";
   423→  const players = _mergedPlayers.map((p) => ({ ...p }));
   424→  if (players.length === 0) {
   425→    view.innerHTML = `
   426→      <div class="dash-empty-state">
   427→        <div class="dash-empty-icon"><i data-icon="chart"></i></div>
   428→        <h3>Aucune donnée disponible</h3>
   429→        <p>${isWeekly ? "Aucune partie cette semaine." : "Chargement…"}</p>
   430→      </div>`;
   431→    if (window.hydrateIcons) window.hydrateIcons(view);
   432→    return;
   433→  }
   434→
   435→  // Ajout du rang
   436→  players.forEach((p, i) => { p.rank = i + 1; });
   437→
   438→  const champion = players[0] ?? null;
   439→  const totalPlayers = players.length;
   440→  const topN = players.slice(0, 50);
   441→  const modeLabel = isWeekly ? "Cette semaine" : "Global";
   442→
   443→  // Indicateur de chargement live
   444→  const liveTag = !_liveFetchDone
   445→    ? `<span class="dash-fallback-tag">⚡ Chargement live des stats… (${_liveFetchProgress}/${_connectedPlayers.length})</span>`
   446→    : _connectedPlayers.length > 0
   447→      ? `<span class="dash-fallback-tag" style="background:rgba(34,197,94,.15);color:#22c55e">✓ Stats live (${_connectedPlayers.length} joueurs connectés)</span>`
   448→      : "";
   449→
   450→  view.innerHTML = `
   451→    ${liveTag}
   452→
   453→    <div class="dash-controls-row">
   454→      <div class="dash-toggle" role="tablist" aria-label="Période du classement">
   455→        <button class="dash-toggle-btn ${!isWeekly ? "active" : ""}" data-mode="global" role="tab" aria-selected="${!isWeekly}">Global</button>
   456→        <button class="dash-toggle-btn ${isWeekly ? "active" : ""}" data-mode="weekly" role="tab" aria-selected="${isWeekly}">Cette semaine</button>
   457→      </div>
   458→      <div class="dash-scoring-inline">FFA casual +10 · FFA classé +1 · Team casual +5 · Team classé +1</div>
   459→    </div>
   460→
   461→    <div class="dash-grid">
   462→      ${champion ? renderChampion(champion, isWeekly) : ""}
   463→      ${renderRanking(topN, totalPlayers, modeLabel)}
   464→    </div>
   465→
   466→    <div class="dash-scoring-info">
   467→      <svg viewBox="0 0 24 24" width="14" height="14" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><circle cx="12" cy="12" r="10"/><line x1="12" y1="16" x2="12" y2="12"/><line x1="12" y1="8" x2="12.01" y2="8"/></svg>
   468→      <span>Barème : <strong>FFA casual +10</strong> · <strong>FFA classé +1</strong> · <strong>Team casual +5</strong> · <strong>Team classé +1</strong> (le classé rapporte juste 1 pt, pas en plus). ${isWeekly ? "Vue hebdomadaire = parties des 7 derniers jours." : "Vue globale = cumul de toutes les parties."} Les stats des joueurs connectés sont récupérées en temps réel via l'API OpenFront.</span>
   469→    </div>
   470→  `;
   471→
   472→  // Hydrate les icônes <i data-icon>
   473→  if (window.hydrateIcons) window.hydrateIcons(view);
   474→
   475→  // Toggle listeners (les boutons sont dans #dashboard-view)
   476→  view.querySelectorAll(".dash-toggle-btn").forEach((btn) => {
   477→    btn.addEventListener("click", () => {
   478→      _dashMode = btn.dataset.mode;
   479→      mergeAndRender();
   480→    });

   740→/* ── Ownership verification (pour le #profile-modal copié de index.html) ── */
   741→
   742→window.startOwnershipVerification = async function () {
   743→  if (!currentUser) {
   744→    showToast("Veuillez vous connecter d'abord.", "warning");
   745→    return;
   746→  }
   747→  const usernameInput = document.getElementById("profile-username");
   748→  const publicIdInput = document.getElementById("profile-public-id");
   749→  const username = (usernameInput?.value || "").trim();
   750→  const publicId = (publicIdInput?.value || "").trim();
   751→
   752→  if (!username || !publicId) {
   753→    showToast("Veuillez remplir tous les champs.", "warning");
   754→    return;
   755→  }
   756→  if (username.length < 2 || username.length > 30) {
   757→    showToast("Le pseudo doit faire entre 2 et 30 caractères.", "warning");
   758→    return;
   759→  }
   760→  if (!/^[A-Za-z0-9]{8}$/.test(publicId)) {
   761→    showToast("Le Public ID doit faire exactement 8 caractères alphanumériques (ex: HabCsQYR).", "warning");
   762→    return;
   763→  }
   764→  if (/[^a-zA-Z0-9_\- ]/.test(username)) {
   765→    showToast("Le pseudo ne peut contenir que des lettres, chiffres, espaces, _ et -.", "warning");
   766→    return;
   767→  }
   768→
   769→  showToast("Vérification du Public ID…", "info", 3000);
   770→  try {
   771→    const playerData = await fetchOpenFront(`/public/player/${encodeURIComponent(publicId)}`);
   772→    if (!playerData || !playerData.publicId) {
   773→      showToast("Public ID introuvable sur OpenFront. Vérifiez votre saisie.", "error");
   774→      return;
   775→    }
   776→  } catch (e) {
   777→    if (e?.isNotFound || e?.status === 404) {
   778→      showToast("Public ID introuvable sur OpenFront. Vérifiez votre saisie.", "error");
   779→      return;
   780→    }
   781→    showToast("Impossible de vérifier le Public ID (API indisponible). Réessayez plus tard.", "error", 6000);
   782→    console.error("[ownership] API check failed:", e);
   783→    return;
   784→  }
   785→
   786→  // Génération du code challenge TFS-XXXX
   787→  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
   788→  _ownershipCode = "TFS-";
   789→  const rand = crypto.getRandomValues(new Uint8Array(4));
   790→  for (let i = 0; i < 4; i++) _ownershipCode += chars[rand[i] % chars.length];
   791→  _ownershipPublicId = publicId;
   792→  _ownershipUsername = username;
   793→
   794→  const s1 = document.getElementById("profile-setup-step1");
   795→  const s2 = document.getElementById("profile-setup-step2");
   796→  if (s1) s1.style.display = "none";
   797→  if (s2) s2.style.display = "block";
   798→  const codeEl = document.getElementById("ownership-code");
   799→  const exEl = document.getElementById("ownership-example");
   800→  if (codeEl) codeEl.textContent = _ownershipCode;
   801→  if (exEl) exEl.textContent = _ownershipCode + " " + username;
   802→  showToast("Code généré. Suivez les instructions ci-dessous.", "info");
   803→  if (window.hydrateIcons) window.hydrateIcons(document.getElementById("profile-modal"));
   804→};

   847→      verified: true,

   849→      createdAt: existing.createdAt || new Date().toISOString(),
   850→      updatedAt: new Date().toISOString(),
