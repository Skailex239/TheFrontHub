    31→// VIP skin: publicId → rewardType (matching par PUBLIC ID, pas par alias)
    32→let vipPlayersByPid = new Map();
    33→let _vipUnsub = null;
    34→const NEW_SKIN_TYPES = ['cyberpunk','sunset','aurore','pastel','gold','volcano','ocean','miami','toxic','chroma','prism'];

    36→/**
    37→ * Écoute public-rewards et construit la map publicId → rewardType.
    38→ * Le skin suit le PUBLIC ID (identité stable) plutôt que l'alias (changeant).
    39→ * On lit data.publicId directement; en fallback on essaie data.username contre
    40→ * le username du profil courant.
    41→ */
    42→function loadVipForProfile() {
    43→  if (_vipUnsub) return; // déjà abonné
    44→  try {
    45→    _vipUnsub = onSnapshot(collection(db, "public-rewards"), (snap) => {
    46→      vipPlayersByPid = new Map();
    47→      // fallback: username → rewardType (pour les docs sans publicId direct)
    48→      const usernameToType = new Map();
    49→      snap.forEach((docSnap) => {
    50→        const data = docSnap.data();
    51→        const rewardType = data.activeType || data.type || null;
    52→        if (!rewardType || data.activated === false) return;
    53→        if (data.publicId) vipPlayersByPid.set(String(data.publicId), rewardType);
    54→        if (data.username) usernameToType.set(data.username, rewardType);
    55→      });
    56→      // Re-applique le skin sur le hero si on a un profil
    57→      // En mode visualisation publique, on utilise le profil virtuel du joueur consulté
    58→      // (son publicId) plutôt que le profil propre de l'utilisateur courant.
    59→      if (viewingPublicId) {
    60→        applyProfileSkin({ username: viewingUsername, publicId: viewingPublicId }, usernameToType);
    61→      } else if (currentProfile) {
    62→        applyProfileSkin(currentProfile, usernameToType);
    63→      }
    64→    }, (err) => {
    65→      console.warn("[profile] VIP listener error (non-critique):", err.message);
    66→    });
    67→  } catch (e) {
    68→    console.warn("[profile] loadVipForProfile error:", e);
    69→  }
    70→}

    72→/**
    73→ * Applique le skin VIP sur le pseudo du hero, résolu via publicId (prioritaire)
    74→ * puis fallback username.
    75→ */
    76→function applyProfileSkin(profile, usernameToTypeFallback) {
    77→  const nameEl = document.getElementById("profile-title-name");
    78→  if (!nameEl) return;
    79→  const pid = profile?.publicId;
    80→  const rewardType = (pid && vipPlayersByPid.get(pid))
    81→    || (profile?.username && usernameToTypeFallback?.get(profile.username))
    82→    || null;
    83→  if (rewardType && NEW_SKIN_TYPES.includes(rewardType)) {
    84→    nameEl.className = `rgb-${rewardType}`;
    85→  } else if (rewardType) {
    86→    nameEl.className = `player-${rewardType}`;
    87→  } else {
    88→    nameEl.className = "";
    89→  }
    90→}

   139→/**
   140→ * Détecte si l'URL demande de visualiser le profil PUBLIC d'un autre joueur.
   141→ * Format: profile.html?player=NAME&publicId=XXXXXXXX
   142→ * Si le publicId correspond à celui de l'utilisateur courant, on ignore
   143→ * (c'est son propre profil — flux normal).
   144→ */
   145→function getPublicProfileRequest() {
   146→  const params = new URLSearchParams(window.location.search);
   147→  const pid = (params.get("publicId") || "").trim();
   148→  const name = (params.get("player") || "").trim();
   149→  if (pid && /^[A-Za-z0-9]{8}$/.test(pid)) {
   150→    return { publicId: pid, username: name || pid };
   151→  }
   152→  return null;
   153→}

   155→onAuthStateChanged(auth, async (user) => {
   156→  // ── Cas 1 : visualisation du profil public d'un autre joueur ──
   157→  // On vérifie l'URL AVANT toute logique d'auth, car cela doit fonctionner
   158→  // même si l'utilisateur n'est pas connecté.
   159→  const pubReq = getPublicProfileRequest();
   160→  if (pubReq) {
   161→    // Lecture du propre profil de l'utilisateur courant (s'il est connecté)
   162→    // pour détecter s'il visualise son PROPRE profil → flux normal.
   163→    let ownProfile = null;
   164→    if (user) {
   165→      try {
   166→        const snap = await getDoc(doc(db, "users", user.uid));
   167→        if (snap.exists()) ownProfile = snap.data();
   168→      } catch (e) {
   169→        console.warn("[profile] Firestore read error (own, non-bloquant):", e.message);
   170→      }
   171→    }
   172→
   173→    if (ownProfile && ownProfile.publicId === pubReq.publicId) {
   174→      // L'utilisateur visualise son propre profil → flux normal (on nettoie l'URL)
   175→      history.replaceState(null, "", window.location.pathname);
   176→      currentUser = user;
   177→      currentProfile = ownProfile;
   178→      updateSidebarUI(user, ownProfile);
   179→      showView("profile-main");
   180→      renderHero(user, ownProfile);
   181→      loadVipForProfile();
   182→      await loadStats(ownProfile.publicId);
   183→      return;
   184→    }
   185→
   186→    // ── Profil d'un AUTRE joueur (ou visiteur non connecté) ──
   187→    currentUser = user; // peut être null
   188→    currentProfile = ownProfile; // pour la sidebar (peut être null)
   189→    updateSidebarUI(user, ownProfile);
   190→    viewingPublicId = pubReq.publicId;
   191→    viewingUsername = pubReq.username;
   192→    showView("profile-main");
   193→    renderPublicProfile(pubReq.username, pubReq.publicId);
   194→    loadVipForProfile();
   195→    await loadStats(pubReq.publicId);
   196→    return;
   197→  }
   198→
   199→  // ── Cas 2 : pas de ?publicId dans l'URL → flux normal ──
   200→  if (!user) {
   201→    currentUser = null;
   202→    currentProfile = null;
   203→    updateSidebarUI(null);
   204→    showView("profile-gate");
   205→    return;
   206→  }
   207→
   208→  currentUser = user;
   209→
   210→  // Read Firestore profile
   211→  let profile = null;
   212→  try {
   213→    const snap = await getDoc(doc(db, "users", user.uid));
   214→    if (snap.exists()) profile = snap.data();
   215→  } catch (e) {
   216→    console.error("[profile] Firestore read error:", e);
   217→    showToast("Erreur de lecture du profil (Firestore).", "error");
   218→  }
   219→
   220→  currentProfile = profile;
   221→  updateSidebarUI(user, profile);
   222→
   223→  if (!profile || !profile.publicId) {
   224→    // New user → setup form
   225→    showView("profile-setup");
   226→    return;
   227→  }
   228→
   229→  // Returning user with publicId → fetch & display stats
   230→  showView("profile-main");
   231→  renderHero(user, profile);
   232→  // Lance l'écoute VIP (skin par publicId) — re-applique le skin dès que les rewards arrivent
   233→  loadVipForProfile();
   234→  await loadStats(profile.publicId);
   235→});

   332→async function loadStats(publicId) {
   333→  // Reset stats list to loading
   334→  setText("stat-week-rank", "This week rank: …");
   335→  setText("stat-week-score", "This week score: …");
   336→  setText("stat-alltime", "All-time score: …");
   337→  const recentEl = document.getElementById("profile-recent-games");
   338→  if (recentEl) recentEl.innerHTML = `<div class="pf-empty">Chargement…</div>`;
   339→  hideError();
   340→
   341→  // Kick off ELO lookup (ranked.json) in parallel
   342→  const eloPromise = getRankedEntry(publicId);
   343→
   344→  // Kick off recent games fetch in parallel (separate endpoint)
   345→  // /public/player/{id} returns aggregated stats only (no games array).
   346→  // /public/player/{id}/games returns the actual recent games list with
   347→  // result (victory/defeat) already included — no need for per-game fetch.
   348→  const recentGamesPromise = fetchRecentGames(publicId);
   349→
   350→  let playerData;
   351→  try {
   352→    playerData = await fetchOpenFront(`/public/player/${encodeURIComponent(publicId)}`);
   353→  } catch (e) {
   354→    console.error("[profile] OpenFront API error:", e);
   355→    showError(`Impossible de charger les statistiques depuis l'API OpenFront.`);
   356→    setText("stat-week-rank", "This week rank: —");
   357→    setText("stat-week-score", "This week score: —");
   358→    setText("stat-alltime", "All-time score: —");
   359→    return;
   360→  }
   361→
   362→  if (!playerData) {
   363→    showError("Réponse vide de l'API OpenFront.");
   364→    return;
   365→  }
   366→
   367→  // NOTE: /public/player/{id} no longer returns a `games` array.
   368→  // Recent games come from the separate /games endpoint (recentGamesPromise).
   369→  const games = [];
   370→  const stats = computeStats(games, playerData.stats || {});
   371→
   372→  // Compute week stats (games in last 7 days)
   373→  const now = Date.now();
   374→  const weekAgo = now - 7 * 24 * 60 * 60 * 1000;
   375→  const weekGames = games.filter(g => {
   376→    const t = new Date(g.start || 0).getTime();
   377→    return !isNaN(t) && t >= weekAgo && t <= now;
   378→  });
   379→  const weekWins = weekGames.length; // approximate: count week games as "score"
   380→  const weekRank = "—"; // not available without global calc
   381→
   382→  // All-time score: wins * 1 + games played (simple heuristic)
   383→  const allTimeScore = stats.wins * 4 + (stats.total - stats.wins);
   384→
   385→  // Breakdown by mode (from stats tree)
   386→  const breakdown = computeModeBreakdown(playerData.stats || {});
   387→  const detail = [];
   388→  if (breakdown.FFA) detail.push("FFA: " + breakdown.FFA);
   389→  if (breakdown.Team) detail.push("Team: " + breakdown.Team);
   390→  if (breakdown.Duos) detail.push("Duos: " + breakdown.Duos);
   391→  if (breakdown.Trios) detail.push("Trios: " + breakdown.Trios);
   392→  if (breakdown.Quads) detail.push("Quads: " + breakdown.Quads);
   393→  const detailStr = detail.length ? " (" + detail.join(", ") + ")" : "";
   394→
   395→  setText("stat-week-rank", `This week rank: #${weekRank}`);
   396→  setText("stat-week-score", `This week score: ${weekWins}`);
   397→  setText("stat-alltime", `All-time score: ${allTimeScore} (${stats.wins} wins${detailStr})`);
   398→
   399→  // ELO from ranked.json (1v1)
   400→  const ranked1v1 = await eloPromise;
   401→  const eloLine = document.getElementById("stat-elo-line");
   402→  if (eloLine) {
   403→    if (ranked1v1 && ranked1v1.elo != null) {
   404→      eloLine.textContent = `ELO 1v1: ${ranked1v1.elo} (Peak: ${ranked1v1.peakElo ?? '—'}) — Rank #${ranked1v1.rank}`;
   405→      eloLine.style.display = "list-item";
   406→    } else {
   407→      eloLine.style.display = "none";
   408→    }
   409→  }
   410→  
   411→  // ELO 2v2 from ranked.json
   412→  const ranked2v2 = await getRankedEntry(publicId, "2v2");
   413→  const elo2v2Line = document.getElementById("stat-elo-2v2-line");
   414→  if (elo2v2Line) {
   415→    if (ranked2v2 && ranked2v2.elo != null) {
   416→      elo2v2Line.textContent = `ELO 2v2: ${ranked2v2.elo} (Peak: ${ranked2v2.peakElo ?? '—'}) — Rank #${ranked2v2.rank}`;
   417→      elo2v2Line.style.display = "list-item";
   418→    } else {
   419→      elo2v2Line.style.display = "none";
   420→    }
   421→  }
   422→
   423→  // Recent games — fetched from /public/player/{id}/games (separate endpoint).
   424→  // The result field (victory/defeat) is already included, no per-game fetch needed.
   425→  try {
   426→    const recentGames = await recentGamesPromise;
   427→    renderRecentGames(recentGames, publicId);
   428→  } catch (e) {
   429→    console.error("[profile] recent games fetch failed:", e);
   430→    const c = document.getElementById("profile-recent-games");
   431→    if (c) c.innerHTML = `<div class="pf-empty">Impossible de charger les parties récentes.</div>`;
   432→  }
   433→}

   435→/**
   436→ * Fetch recent games for a player from the /public/player/{id}/games endpoint.
   437→ * Returns up to 10 games with result (victory/defeat) already included.
   438→ * Supports cursor pagination to fetch more if needed.
   439→ */
   440→async function fetchRecentGames(publicId, maxPages = 1) {
   441→  const all = [];
   442→  let cursor = null;
   443→  for (let page = 0; page < maxPages; page++) {
   444→    const url = `/public/player/${encodeURIComponent(publicId)}/games` +
   445→      (cursor ? `?cursor=${encodeURIComponent(cursor)}` : "");
   446→    const data = await fetchOpenFront(url);
   447→    const results = Array.isArray(data?.results) ? data.results : [];
   448→    all.push(...results);
   449→    cursor = data?.nextCursor;
   450→    if (!cursor || results.length === 0) break;
   451→  }
   452→  return all;
   453→}

   460→function computeModeBreakdown(statsTree) {
   461→  const out = { FFA: 0, Team: 0, Duos: 0, Trios: 0, Quads: 0 };
   462→  if (!statsTree || typeof statsTree !== "object") return out;
   463→  for (const catKey of Object.keys(statsTree)) {
   464→    const cat = statsTree[catKey];
   465→    if (!cat || typeof cat !== "object") continue;
   466→    for (const modeKey of Object.keys(cat)) {
   467→      const mode = cat[modeKey];
   468→      if (!mode || typeof mode !== "object") continue;
   469→      let wins = 0;
   470→      for (const diffKey of Object.keys(mode)) {
   471→        const diff = mode[diffKey];
   472→        if (diff && typeof diff === "object" && diff.wins != null) {
   473→          wins += parseInt(diff.wins, 10) || 0;
   474→        }
   475→      }
   476→      if (modeKey === "Free For All") out.FFA += wins;
   477→      else if (modeKey === "Team") {
   478→        // Try to break down by playerTeams if available
   479→        out.Team += wins;
   480→      }
   481→    }
   482→  }
   483→  return out;
   484→}

   486→function computeStats(games, statsTree) {
   487→  // Wins: sum all "wins" fields across the stats tree (Private/Public/Ranked → mode → difficulty)
   488→  let wins = 0;
   489→  let total = 0;
   490→  if (statsTree && typeof statsTree === "object") {
   491→    for (const catKey of Object.keys(statsTree)) {
   492→      const cat = statsTree[catKey];
   493→      if (!cat || typeof cat !== "object") continue;
   494→      for (const modeKey of Object.keys(cat)) {
   495→        const mode = cat[modeKey];
   496→        if (!mode || typeof mode !== "object") continue;
   497→        for (const diffKey of Object.keys(mode)) {
   498→          const diff = mode[diffKey];
   499→          if (!diff || typeof diff !== "object") continue;
   500→          if (diff.wins != null) wins += parseInt(diff.wins, 10) || 0;
   501→          if (diff.total != null) total += parseInt(diff.total, 10) || 0;
   502→          else if (diff.wins != null && diff.losses != null) {
   503→            total += (parseInt(diff.wins, 10) || 0) + (parseInt(diff.losses, 10) || 0);
   504→          }
   505→        }
   506→      }
   507→    }
   508→  }
   509→
   510→  // Fallback: if stats tree has no totals, use games.length
   511→  if (total === 0 && games.length > 0) total = games.length;
   512→
   513→  // Unique maps + favourite map
   514→  const mapCounts = {};
   515→  let lastGame = null;
   516→  for (const g of games) {
   517→    if (g.map) mapCounts[g.map] = (mapCounts[g.map] || 0) + 1;
   518→    if (g.start) {
   519→      const d = new Date(g.start).getTime();
   520→      if (!isNaN(d) && (lastGame === null || d > lastGame)) lastGame = d;
   521→    }
   522→  }
   523→  const uniqueMaps = Object.keys(mapCounts).length;
   524→  let favMap = null;
   525→  let favCount = 0;
   526→  for (const [m, c] of Object.entries(mapCounts)) {
   527→    if (c > favCount) { favMap = m; favCount = c; }
   528→  }
   529→  const lastGameIso = lastGame ? new Date(lastGame).toISOString() : null;
   530→
   531→  return { wins, total, uniqueMaps, favMap, lastGame: lastGameIso };
   532→}

   562→/**
   563→ * Render recent games from /public/player/{id}/games endpoint.
   564→ * Each game already includes a `result` field ("victory" | "defeat" | other)
   565→ * so no per-game fetch is needed.
   566→ *
   567→ * Game object structure:
   568→ *   { gameId, start, durationSeconds, map, mode, type, playerTeams,
   569→ *     rankedType, result, totalPlayers, username, clanTag }
   570→ */
   571→function renderRecentGames(games, publicId) {
   572→  const container = document.getElementById("profile-recent-games");
   573→  if (!container) return;
   574→
   575→  // Sort by start date desc, take last 10 (API returns 10 per page)
   576→  const sorted = games
   577→    .slice()
   578→    .sort((a, b) => new Date(b.start || 0).getTime() - new Date(a.start || 0).getTime())
   579→    .slice(0, 10);
   580→
   581→  if (sorted.length === 0) {
   582→    container.innerHTML = `<div class="pf-empty">Aucune partie récente.</div>`;
   583→    return;
   584→  }
   585→
   586→  container.innerHTML = sorted.map((g) => {
   587→    const isWin = g.result === "victory";
   588→    const resultClass = isWin ? "win" : "loss";
   589→    const resultLabel = isWin ? "VICTOIRE" : (g.result === "defeat" ? "DÉFAITE" : (g.result || "—"));
   590→    const duration = g.durationSeconds ? formatDuration(g.durationSeconds) : "—";
   591→    const modeLabel = formatGameMode(g);
   592→    const mapName = g.map || "Carte inconnue";
   593→    const rankedBadge = g.rankedType && g.rankedType !== "unranked"
   594→      ? `<span class="pf-game-ranked">${esc(g.rankedType)}</span>` : "";
   595→    const totalPlayers = g.totalPlayers != null ? `${g.totalPlayers} joueurs` : "";
   596→
   597→    return `
   598→      <div class="pf-game-card ${resultClass}">
   599→        <div class="pf-game-result">${resultLabel}</div>
   600→        <div class="pf-game-info">
   601→          <div class="pf-game-map">${esc(mapName)} ${rankedBadge}</div>
   602→          <div class="pf-game-meta">${esc(modeLabel)}${totalPlayers ? ' · ' + esc(totalPlayers) : ''} · ${duration}</div>
   603→          <div class="pf-game-meta">${formatDateTime(g.start)}</div>
   604→        </div>
   605→        <a class="pf-game-replay" href="https://openfront.io/game/${encodeURIComponent(g.gameId)}" target="_blank" rel="noopener" title="Voir le replay">
   606→          <svg viewBox="0 0 24 24" width="14" height="14" fill="currentColor"><polygon points="5 3 19 12 5 21 5 3"/></svg>
   607→        </a>
   608→      </div>
   609→    `;
   610→  }).join("");
   611→}

   613→/** Format duration in seconds as M:SS or H:MM:SS */
   614→function formatDuration(seconds) {
   615→  const s = Math.floor(Number(seconds) || 0);
   616→  if (s < 60) return `${s}s`;
   617→  const m = Math.floor(s / 60);
   618→  const rs = s % 60;
   619→  if (m < 60) return `${m}:${String(rs).padStart(2, "0")}`;
   620→  const h = Math.floor(m / 60);
   621→  const rm = m % 60;
   622→  return `${h}:${String(rm).padStart(2, "0")}:${String(rs).padStart(2, "0")}`;
   623→}

   625→/** Build a human-readable game mode label from the game object */
   626→function formatGameMode(g) {
   627→  const parts = [];
   628→  if (g.type) parts.push(g.type);
   629→  if (g.mode) parts.push(g.mode === "Free For All" ? "FFA" : g.mode);
   630→  if (g.playerTeams && g.playerTeams !== "null") parts.push(g.playerTeams);
   631→  return parts.join(" · ") || "—";
   632→}

   634→/**
   635→ * Check whether the given clientId is among the winners of the given game.
   636→ * OpenFront `/public/game/{gameId}` returns `info.winner` as
   637→ * `[type, name, ...clientIDs]`.
   638→ */
   639→async function checkGameWin(gameId, clientId) {
   640→  if (!gameId || !clientId) return null;
   641→  const data = await fetchOpenFront(`/public/game/${encodeURIComponent(gameId)}`);
   642→  const winner = data?.info?.winner;
   643→  if (!Array.isArray(winner) || winner.length < 3) return null;
   644→  // winner[0] = "team" | "player", winner[1] = name, winner[2..] = clientIDs
   645→  const winnerIds = winner.slice(2);
   646→  return winnerIds.includes(clientId);
   647→}

   666→    showToast("Le pseudo doit faire entre 2 et 30 caractères.", "warning");
   667→    return;
   668→  }
   669→  if (!/^[A-Za-z0-9]{8}$/.test(publicId)) {

   674→    showToast("Le pseudo ne peut contenir que des lettres, chiffres, espaces, _ et -.", "warning");
   675→    return;

   681→      showToast("Le Public ID OpenFront ne peut plus être modifié.", "error");
   682→      return;
   683→    }
   684→  } catch (e) { /* non-blocking */ }
   685→
   686→  // Verify publicId exists on OpenFront
   687→  showToast("Vérification du Public ID…", "info", 3000);
   688→  try {
   689→    const playerData = await fetchOpenFront(`/public/player/${encodeURIComponent(publicId)}`);

   696→    console.error("[ownership] API check failed:", e);
   697→    return;
   698→  }
   699→
   700→  // Generate TFS-XXXX challenge code
   701→  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
   702→  _ownershipCode = "TFS-";
   703→  const _randBytes = crypto.getRandomValues(new Uint8Array(4));
   704→  for (let i = 0; i < 4; i++) _ownershipCode += chars[_randBytes[i] % chars.length];

   755→async function saveUserProfile(username, publicId) {
   756→  if (!currentUser) throw new Error("No authenticated user");
   757→  try {
   758→    const existing = currentProfile || {};
   759→    await setDoc(doc(db, "users", currentUser.uid), {
   760→      username,
   761→      publicId,
   762→      email: currentUser.email || null,
   763→      verified: true,
   764→      verifiedAt: new Date().toISOString(),
   765→      createdAt: existing.createdAt || new Date().toISOString(),
   766→      updatedAt: new Date().toISOString(),
   767→    }, { merge: true });
   768→
   769→    currentProfile = { ...(currentProfile || {}), username, publicId, verified: true };
   770→    showToast("Profil vérifié et enregistré avec succès !", "success");
   771→
   772→    // Publie le lien publicId ↔ username/uid dans une collection publique pour que
   773→    // le matching VIP par PUBLIC ID fonctionne pour tous les viewers (skin suit le
   774→    // public_id, pas l'alias). Best-effort: ignoré silencieusement si règles bloquent.
   775→    try {
   776→      await setDoc(doc(db, "public-aliases", currentUser.uid), {
   777→        username,
   778→        publicId,
   779→        aliases: [username],
   780→        updatedAt: new Date().toISOString(),
   781→      }, { merge: true });
   782→    } catch (bridgeErr) {
   783→      console.warn("[profile] Bridge public-aliases (publicId) write failed (non-critique):", bridgeErr.message);
   784→    }
   785→    try {
   786→      await setDoc(doc(db, "public-rewards", currentUser.uid), {
   787→        publicId,
   788→        username,
   789→      }, { merge: true });
   790→    } catch (rewardsErr) {
   791→      console.warn("[profile] public-rewards publicId merge failed (non-critique):", rewardsErr.message);
   792→    }
   793→
   794→    // Reset setup form
   795→    window.cancelOwnershipVerification();
   796→    updateSidebarUI(currentUser, currentProfile);
   797→
   798→    // Switch to main view and load stats
   799→    showView("profile-main");
   800→    renderHero(currentUser, currentProfile);
   801→    loadVipForProfile(); // écoute VIP pour appliquer le skin par publicId
   802→    await loadStats(publicId);
   803→  } catch (e) {
   804→    console.error("[profile] Save profile error:", e);
   805→    showToast("Erreur lors de la sauvegarde du profil.", "error");
   806→    throw e;
   807→  }
   808→}
