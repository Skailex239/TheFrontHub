    66→    66→function escapeHtml(s) {
    67→    67→  if (s == null) return "";
    68→    68→  return String(s)
    69→    69→    .replace(/&/g, "&amp;")
    70→    70→    .replace(/</g, "&lt;")
    71→    71→    .replace(/>/g, "&gt;")
    72→    72→    .replace(/"/g, "&quot;")
    73→    73→    .replace(/'/g, "&#39;");
    74→    74→}

    80→    80→function initials(name) {
    81→    81→  if (!name) return "?";
    82→    82→  const clean = name.replace(/^\[[^\]]+\]\s*/, "").trim();
    83→    83→  const parts = clean.split(/[\s_-]+/).filter(Boolean);
    84→    84→  const letters = parts.length >= 2 ? parts[0][0] + parts[1][0] : clean.slice(0, 2);
    85→    85→  return letters.toUpperCase();
    86→    86→}

   114→   114→async function loadData() {
   115→   115→  // Tentative principale : dashboard_ranking.json (cache-bust)
   116→   116→  try {
   117→   117→    const res = await fetch(`dashboard_ranking.json?v=${Date.now()}`, { cache: "no-store" });
   118→   118→    if (res.ok) {
   119→   119→      const json = await res.json();
   120→   120→      if (json && (json.global?.players?.length || json.weekly?.players?.length)) {
   121→   121→        _data = json;
   122→   122→        if (_data.updatedAt) updateLastUpdateLabel(_data.updatedAt);
   123→   123→        return;
   124→   124→      }
   125→   125→    }
   126→   126→  } catch (e) {
   127→   127→    console.warn("[dashboard] dashboard_ranking.json indisponible:", e.message);
   128→   128→  }
   129→   129→
   130→   130→  // Fallback : ranked.json (top 100 classé 1v1 + 2v2)
   131→   131→  _data = null;
   132→   132→  try {
   133→   133→    const res = await fetch("ranked.json", { cache: "no-store" });
   134→   134→    if (res.ok) _fallbackRanked = await res.json();
   135→   135→  } catch (e) {
   136→   136→    console.warn("[dashboard] ranked.json indisponible:", e.message);
   137→   137→  }
   138→   138→  if (_fallbackRanked?.updatedAt) updateLastUpdateLabel(_fallbackRanked.updatedAt);
   139→   139→}

   160→   160→function pointsFor(p) {
   161→   161→  // Le JSON fournit déjà `points`, mais on le recalcule pour être sûr
   162→   162→  // de la cohérence avec le barème affiché.
   163→   163→  return (p.ffaCasualWins || 0) * PTS_FFA_CASUAL
   164→   164→       + (p.ffaRankedWins || 0) * PTS_FFA_RANKED
   165→   165→       + (p.teamCasualWins || 0) * PTS_TEAM_CASUAL
   166→   166→       + (p.teamRankedWins || 0) * PTS_TEAM_RANKED;
   167→   167→}

   169→   169→function rankedView() {
   170→   170→  // Construit une vue synthétique à partir du fallback ranked.json
   171→   171→  // (casual = 0 partout, ranked wins = wins du leaderboard).
   172→   172→  if (!_fallbackRanked) return null;
   173→   173→  const byPid = new Map();
   174→   174→  const getOrCreate = (pid, name) => {
   175→   175→    let e = byPid.get(pid);
   176→   176→    if (!e) {
   177→   177→      e = {
   178→   178→        publicId: pid, username: name || pid, clan: null,
   179→   179→        ffaCasualWins: 0, ffaRankedWins: 0,
   180→   180→        teamCasualWins: 0, teamRankedWins: 0,
   181→   181→      };
   182→   182→      byPid.set(pid, e);
   183→   183→    }
   184→   184→    return e;
   185→   185→  };
   186→   186→  for (const p of _fallbackRanked["1v1"] || []) {
   187→   187→    const nm = p.username || p.accountUsername || p.public_id;
   188→   188→    const e = getOrCreate(p.public_id, nm);
   189→   189→    e.ffaRankedWins = p.wins || 0;
   190→   190→    if (nm && nm !== p.public_id) e.username = nm;
   191→   191→  }
   192→   192→  for (const p of _fallbackRanked["2v2"] || []) {
   193→   193→    const nm = p.username || p.accountUsername || p.public_id;
   194→   194→    const e = getOrCreate(p.public_id, nm);
   195→   195→    e.teamRankedWins = p.wins || 0;
   196→   196→    if (nm && nm !== p.public_id) e.username = nm;
   197→   197→  }
   198→   198→  const players = [...byPid.values()].map((p) => ({ ...p, points: pointsFor(p) }));
   199→   199→  players.sort((a, b) => b.points - a.points);
   200→   200→  return {
   201→   201→    from: null, to: null,
   202→   202→    gamesScanned: null,
   203→   203→    players,
   204→   204→    _fallback: true,
   205→   205→  };
   206→   206→}

   580→   580→window.startOwnershipVerification = async function () {
   581→   581→  if (!currentUser) {

   583→   583→    return;
   584→   584→  }
   585→   585→  const usernameInput = document.getElementById("profile-username");
   586→   586→  const publicIdInput = document.getElementById("profile-public-id");
   587→   587→  const username = (usernameInput?.value || "").trim();
   588→   588→  const publicId = (publicIdInput?.value || "").trim();
   589→   589→
   590→   590→  if (!username || !publicId) {
   591→   591→    showToast("Veuillez remplir tous les champs.", "warning");
   592→   592→    return;
   593→   593→  }
   594→   594→  if (username.length < 2 || username.length > 30) {
   595→   595→    showToast("Le pseudo doit faire entre 2 et 30 caractères.", "warning");
   596→   596→    return;
   597→   597→  }
   598→   598→  if (!/^[A-Za-z0-9]{8}$/.test(publicId)) {
   599→   599→    showToast("Le Public ID doit faire exactement 8 caractères alphanumériques (ex: HabCsQYR).", "warning");
   600→   600→    return;
   601→   601→  }
   602→   602→  if (/[^a-zA-Z0-9_\- ]/.test(username)) {
   603→   603→    showToast("Le pseudo ne peut contenir que des lettres, chiffres, espaces, _ et -.", "warning");
   604→   604→    return;
   605→   605→  }
   606→   606→
   607→   607→  showToast("Vérification du Public ID…", "info", 3000);
   608→   608→  try {
   609→   609→    const playerData = await fetchOpenFront(`/public/player/${encodeURIComponent(publicId)}`);
   610→   610→    if (!playerData || !playerData.publicId) {
   611→   611→      showToast("Public ID introuvable sur OpenFront. Vérifiez votre saisie.", "error");
   612→   612→      return;
   613→   613→    }
   614→   614→  } catch (e) {
   615→   615→    if (e?.isNotFound || e?.status === 404) {
   616→   616→      showToast("Public ID introuvable sur OpenFront. Vérifiez votre saisie.", "error");
   617→   617→      return;
   618→   618→    }
   619→   619→    showToast("Impossible de vérifier le Public ID (API indisponible). Réessayez plus tard.", "error", 6000);
   620→   620→    console.error("[ownership] API check failed:", e);
   621→   621→    return;
   622→   622→  }
   623→   623→
   624→   624→  // Génération du code challenge TFS-XXXX
   625→   625→  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
   626→   626→  _ownershipCode = "TFS-";
   627→   627→  const rand = crypto.getRandomValues(new Uint8Array(4));
   628→   628→  for (let i = 0; i < 4; i++) _ownershipCode += chars[rand[i] % chars.length];
   629→   629→  _ownershipPublicId = publicId;
   630→   630→  _ownershipUsername = username;
   631→   631→
   632→   632→  const s1 = document.getElementById("profile-setup-step1");

   638→   638→  if (codeEl) codeEl.textContent = _ownershipCode;
   639→   639→  if (exEl) exEl.textContent = _ownershipCode + " " + username;
   640→   640→  showToast("Code généré. Suivez les instructions ci-dessous.", "info");
   641→   641→  if (window.hydrateIcons) window.hydrateIcons(document.getElementById("profile-modal"));
   642→   642→};

   676→   676→async function saveUserProfile(username, publicId) {

   685→   685→      verified: true,
   686→   686→      verifiedAt: new Date().toISOString(),
   687→   687→      createdAt: existing.createdAt || new Date().toISOString(),
   688→   688→      updatedAt: new Date().toISOString(),
   689→   689→      openFrontSyncPending: true,
   690→   690→    }, { merge: true });
   691→   691→
   692→   692→    currentUser.name = username;
