   235→   235→// ── Fetch avec retry et gestion 429 ──────────────────────────────────────────
   236→   236→async function fetchWithRetry(url, retries = 3) {
   237→   237→  for (let attempt = 0; attempt <= retries; attempt++) {
   238→   238→    const controller = new AbortController();
   239→   239→    const timer = setTimeout(() => controller.abort(), FETCH_TIMEOUT);
   240→   240→    try {
   241→   241→      const res = await openFrontFetch(url, { signal: controller.signal });
   242→   242→      clearTimeout(timer);
   243→   243→      if (res.status === 429) {
   244→   244→        const wait = DELAY_429 * (attempt + 1);
   245→   245→        console.log(`[rate-limit] 429 — attente ${wait}ms (tentative ${attempt + 1}/${retries})`);
   246→   246→        await sleep(wait);
   247→   247→        continue;
   248→   248→      }
   249→   249→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   250→   250→      return await res.json();
   251→   251→    } catch (e) {
   252→   252→      clearTimeout(timer);
   253→   253→      if (e.name === "AbortError") {
   254→   254→        if (attempt < retries) { await sleep(500); continue; }
   255→   255→        throw new Error("Timeout");
   256→   256→      }
   257→   257→      if (attempt === retries) throw e;
   258→   258→      await sleep(500 * (attempt + 1));
   259→   259→    }
   260→   260→  }
   261→   261→}

   325→   325→async function processGames(games, seen, runs, runIds) {
   326→   326→  const unseen = games.filter(g => g.game && !seen.has(g.game));
   327→   327→  if (unseen.length === 0) return 0;
   328→   328→
   329→   329→  let newRuns = 0;
   330→   330→  let errors = 0;
   331→   331→
   332→   332→  console.log(
   333→   333→    `[sync] ${unseen.length} parties à détailler (×${DETAIL_CONCURRENCY} parallèle${HAS_EXEMPTION ? ", mode rapide" : ""})`
   334→   334→  );
   335→   335→
   336→   336→  for (let i = 0; i < unseen.length; i += DETAIL_CONCURRENCY) {
   337→   337→    const chunk = unseen.slice(i, i + DETAIL_CONCURRENCY);
   338→   338→    const results = await Promise.all(
   339→   339→      chunk.map(game => processOneGame(game, seen, runs, runIds))
   340→   340→    );
   341→   341→    for (const r of results) {
   342→   342→      if (typeof r === "number") newRuns += r;
   343→   343→      else if (r?.error) {
   344→   344→        errors++;
   345→   345→        if (errors <= 5) console.warn(`[sync] ⚠️ ${r.gameId}: ${r.error.message}`);
   346→   346→      }
   347→   347→    }
   348→   348→    if (BATCH_DELAY_NORMAL > 0) await sleep(BATCH_DELAY_NORMAL);
   349→   349→  }
   350→   350→
   351→   351→  if (errors > 5) console.log(`[sync] ... et ${errors - 5} autres erreurs`);
   352→   352→  return newRuns;
   353→   353→}

   360→   360→async function syncRecent() {
   361→   361→  console.log(`[sync] 🔄 Sync récente — ${new Date().toISOString()}`);
   362→   362→  const seen = loadSeen();
   363→   363→  const runs = loadRuns();
   364→   364→  const runIds = new Set(runs.map(r => r.id));
   365→   365→  let totalNew = 0;
   366→   366→
   367→   367→  const now = new Date();
   368→   368→  const cp = loadCheckpoints();
   369→   369→  const lastSync = cp.last_sync_time ? parseInt(cp.last_sync_time, 10) : 0;
   370→   370→  const agoMs = Math.max(now.getTime() - RECENT_MAX_MS, lastSync - RECENT_OVERLAP_MS);
   371→   371→  const ago = new Date(agoMs);
   372→   372→  const windowMin = Math.round((now - ago) / 60_000);
   373→   373→
   374→   374→  const windows = buildWindows30s(ago, now);
   375→   375→  console.log(
   376→   376→    `[sync] ${windows.length} fenêtres de 30s (~${windowMin} min, max 3h, filtre Public FFA ≥10p)`
   377→   377→  );
   378→   378→
   379→   379→  for (const { start, end } of windows) {
   380→   380→    const games = await fetchGamesInWindow(start, end);
   381→   381→    if (games.length > 0) {
   382→   382→      if (games.length >= WINDOW_SATURATION_THRESHOLD) {
   383→   383→        console.log(
   384→   384→          `[sync] ⚠️ Fenêtre saturée (${games.length}) ${start.toISOString().slice(11, 19)} — possible troncature`
   385→   385→        );
   386→   386→      }
   387→   387→      totalNew += await processGames(games, seen, runs, runIds);
   388→   388→    }
   389→   389→    if (WINDOW_DELAY > 0) await sleep(WINDOW_DELAY);
   390→   390→  }
   391→   391→
   392→   392→  if (totalNew > 0) saveRuns(runs);
   393→   393→  saveSeen(seen);
   394→   394→
   395→   395→  cp.last_sync_time = String(Date.now());
   396→   396→  saveCheckpoints(cp);
   397→   397→
   398→   398→  logApiStats("sync-recent");
   399→   399→  console.log(`[sync] ✅ Sync récente terminée — ${totalNew} nouveaux runs`);
   400→   400→  return totalNew;
   401→   401→}

   461→   461→async function syncHistory(maxWindows = DEFAULT_HISTORY_WINDOWS) {
   462→   462→  const cp = loadCheckpoints();
   463→   463→  const oldest = TARGET_DATE;
   464→   464→  const now = Date.now();
   465→   465→
   466→   466→  const saved = cp.history_oldest_reached;
   467→   467→  const resumeFrom = saved ? Math.max(parseInt(saved) - WINDOW_MS, oldest) : now;
   468→   468→
   469→   469→  printSyncStatus(cp);
   470→   470→
   471→   471→  if (parseInt(saved) <= oldest + WINDOW_MS * 2) {
   472→   472→    console.log(`[history] ✅ Historique complet jusqu'au ${new Date(oldest).toISOString().slice(0, 10)}`);
   473→   473→    console.log("[history] Utilise reset-history puis history 500 pour rescanner les trous.");
   474→   474→    return 0;
   475→   475→  }
   476→   476→
   477→   477→  console.log(`[history] 🕐 Reprise depuis ${new Date(resumeFrom).toISOString()}`);
   478→   478→
   479→   479→  const rangeEnd = new Date(resumeFrom);
   480→   480→  const rangeStart = new Date(oldest);
   481→   481→  const windows = buildWindows30s(rangeStart, rangeEnd);
   482→   482→
   483→   483→  const toProcess = Math.min(windows.length, maxWindows);
   484→   484→  console.log(`[history] ${windows.length.toLocaleString()} fenêtres restantes — traitement de ${toProcess} (max ${maxWindows})`);
   485→   485→
   486→   486→  const seen = loadSeen();
   487→   487→  const runs = loadRuns();
   488→   488→  const runIds = new Set(runs.map(r => r.id));
   489→   489→  let totalRuns = 0;
   490→   490→  let oldestReached = resumeFrom;
   491→   491→  let saturatedWindows = 0;
   492→   492→
   493→   493→  for (let i = 0; i < toProcess; i++) {
   494→   494→    const { start, end } = windows[i];
   495→   495→    try {
   496→   496→      const games = await fetchGamesInWindow(start, end);
   497→   497→      if (games.length > 0) {
   498→   498→        if (games.length >= WINDOW_SATURATION_THRESHOLD) {
   499→   499→          saturatedWindows++;
   500→   500→          if (saturatedWindows <= 3) {
   501→   501→            console.log(
   502→   502→              `[history] ⚠️ Fenêtre saturée (${games.length}): ${start.toISOString().slice(0, 16)}`
   503→   503→            );
   504→   504→          }
   505→   505→        }
   506→   506→        const added = await processGames(games, seen, runs, runIds);
   507→   507→        totalRuns += added;
   508→   508→        if (added > 0) {
   509→   509→          console.log(`[history] +${added} runs (${start.toISOString().slice(0, 10)} ${start.toISOString().slice(11, 16)})`);
   510→   510→        }
   511→   511→      }
   512→   512→      oldestReached = end.getTime();
   513→   513→    } catch (e) {
   514→   514→      console.warn(`[history] ⚠️ Erreur fenêtre ${start.toISOString()}: ${e.message}`);
   515→   515→    }
   516→   516→
   517→   517→    if (WINDOW_DELAY > 0) await sleep(WINDOW_DELAY);
   518→   518→
   519→   519→    // Checkpoint périodique
   520→   520→    if ((i + 1) % CHECKPOINT_EVERY === 0 || i === toProcess - 1) {
   521→   521→      cp.history_oldest_reached = String(oldestReached);
   522→   522→      cp.history_saturated_windows = (cp.history_saturated_windows || 0) + saturatedWindows;
   523→   523→      saveCheckpoints(cp);
   524→   524→      saveSeen(seen);
   525→   525→      if (totalRuns > 0) saveRuns(runs);
   526→   526→
   527→   527→      const pct = Math.round(((now - oldestReached) / (now - oldest)) * 100);
   528→   528→      console.log(`[history] 💾 ${i + 1}/${toProcess} fenêtres — ${totalRuns} runs — ${pct}% de l'historique`);
   529→   529→    }
   530→   530→  }
   531→   531→
   532→   532→  // Sauvegarde finale
   533→   533→  if (totalRuns > 0) saveRuns(runs);
   534→   534→  saveSeen(seen);
   535→   535→
   536→   536→  if (saturatedWindows > 0) {
   537→   537→    console.log(`[history] ⚠️ ${saturatedWindows} fenêtres saturées détectées — certains runs ont peut-être été manqués`);
   538→   538→  }
   539→   539→
   540→   540→  if (windows.length > maxWindows) {
   541→   541→    console.log(`[history] ⏹️ Limite atteinte — reprendra au prochain run (reste: ${(windows.length - toProcess).toLocaleString()} fenêtres)`);
   542→   542→  } else {
   543→   543→    console.log(`[history] ✅ Historique terminé — ${totalRuns} runs insérés`);
   544→   544→  }
   545→   545→
   546→   546→  return totalRuns;
   547→   547→}
