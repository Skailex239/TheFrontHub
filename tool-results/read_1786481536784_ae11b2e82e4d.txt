     1→/**
     2→ * dashboard.js — Contrôleur du Tableau de bord TheFrontHub.
     3→ *
     4→ * Architecture (NO SYNC — le site fait les requêtes API lui-même) :
     5→ *
     6→ *   1. ranked.json (auto-synced par GitHub Actions) → ranked wins carrière
     7→ *      pour le top 100 1v1 + top 100 2v2. Servé depuis GitHub Pages.
     8→ *
     9→ *   2. Firebase public-aliases (Firestore REST API) → liste des joueurs
    10→ *      connectés via Google/Discord qui ont lié leur Public ID.
    11→ *
    12→ *   3. OpenFront API (via proxy Cloudflare Worker ou /api/openfront en dev)
    13→ *      → pour chaque joueur connecté, on pagine TOUTES ses parties et on
    14→ *      compte les victoires par catégorie (FFA casual, FFA classé, Team
    15→ *      casual, Team classé). Le proxy ajoute le header x-skailex-access
    16→ *      côté serveur (exemption de rate-limit).
    17→ *
    18→ *   Merge :
    19→ *     - Joueurs ranked-only : ranked wins de ranked.json, casual = 0
    20→ *     - Joueurs connectés : ranked wins = max(ranked.json, API carrière),
    21→ *       casual wins = API live
    22→ *
    23→ *   Cache : les stats live sont mises en cache dans localStorage (30 min)
    24→ *   pour éviter de re-fetcher à chaque visite.
    25→ *
    26→ * Barème :
    27→ *   FFA casual  = +10  ·  FFA classé (1v1) = +1
    28→ *   Team casual = +5   ·  Team classé (2v2) = +1
    29→ *   (ranked = 1 pt, PAS en plus du FFA/Team)
    30→ *
    31→ * Auth : importe auth.js (Firebase) et écoute onAuthStateChanged pour brancher
    32→ * la sidebar (login-btn / user-badge). Définit sur window les handlers utilisés
    33→ * par les onclick du HTML : toggleAuthModal, handleLogin, handleLogout,
    34→ * toggleUserDropdown, goToProfilePage, closeProfileModal,
    35→ * startOwnershipVerification, confirmOwnershipVerification,
    36→ * cancelOwnershipVerification.
    37→ */

    90→  if (s == null) return "";
    91→  return String(s)
    92→    .replace(/&/g, "&amp;")
    93→    .replace(/</g, "&lt;")
    94→    .replace(/>/g, "&gt;")
    95→    .replace(/"/g, "&quot;")
    96→    .replace(/'/g, "&#39;");

   103→/**
   104→ * Retourne le timestamp (ms) du lundi 00:00 (heure locale navigateur)
   105→ * de la semaine contenant `now`. Utilisé pour le label "Depuis lundi …".
   106→ */
   107→function getWeekStartMs(now) {
   108→  const d = new Date(now);
   109→  d.setHours(0, 0, 0, 0);
   110→  // getDay(): 0=Sunday, 1=Monday... On veut lundi comme début de semaine
   111→  const day = d.getDay();
   112→  const diff = day === 0 ? -6 : 1 - day; // recule jusqu'à lundi
   113→  d.setDate(d.getDate() + diff);
   114→  return d.getTime();
   115→}

   127→function initials(name) {
   128→  if (!name) return "?";
   129→  const clean = name.replace(/^\[[^\]]+\]\s*/, "").trim();
   130→  const parts = clean.split(/[\s_-]+/).filter(Boolean);
   131→  const letters = parts.length >= 2 ? parts[0][0] + parts[1][0] : clean.slice(0, 2);
   132→  return letters.toUpperCase();
   133→}

   175→/** Charge la liste des joueurs connectés depuis Firebase public-aliases. */
   176→async function loadConnectedPlayers() {
   177→  try {
   178→    const url = `${FIRESTORE_BASE}/public-aliases`;
   179→    const res = await fetch(url, { cache: "no-store" });
   180→    if (!res.ok) {
   181→      console.warn(`[dashboard] Firebase public-aliases: HTTP ${res.status}`);
   182→      return;
   183→    }
   184→    const data = await res.json();
   185→    const docs = data.documents || [];
   186→    const seen = new Set();
   187→    _connectedPlayers = docs.map((doc) => {
   188→      const fields = doc.fields || {};
   189→      const val = (f) => (f?.stringValue || f?.integerValue || "");
   190→      const publicId = val(fields.publicId);
   191→      return {
   192→        publicId,
   193→        username: val(fields.username) || publicId || "?",
   194→      };
   195→    })
   196→    // Filtrer : publicId valide = exactement 8 caractères alphanumériques
   197→    .filter((p) => /^[A-Za-z0-9]{8}$/.test(p.publicId))
   198→    // Dédoublonner par publicId (garder le premier)
   199→    .filter((p) => {
   200→      if (seen.has(p.publicId)) return false;
   201→      seen.add(p.publicId);
   202→      return true;
   203→    });
   204→    console.log(`[dashboard] Firebase: ${_connectedPlayers.length} joueurs connectés (sur ${docs.length} documents)`);
   205→  } catch (e) {
   206→    console.warn("[dashboard] Firebase indisponible:", e.message);
   207→  }
   208→}

   210→/**
   211→ * Pagine TOUTES les parties d'un joueur via le proxy OpenFront.
   212→ * Retourne [{ gameId, start, mode, rankedType, result, map, ... }].
   213→ */
   214→async function fetchAllPlayerGames(publicId) {
   215→  const allGames = [];
   216→  let cursor = null;
   217→  for (let page = 0; page < 500; page++) {
   218→    let apiPath = `/public/player/${encodeURIComponent(publicId)}/games`;
   219→    if (cursor) apiPath += `?cursor=${encodeURIComponent(cursor)}`;
   220→    let data;
   221→    try {
   222→      data = await fetchOpenFront(apiPath);
   223→    } catch (e) {
   224→      console.warn(`[dashboard] fetch games ${publicId} page ${page}:`, e.message);
   225→      break;
   226→    }
   227→    const results = data?.results || [];
   228→    if (results.length === 0) break;
   229→    allGames.push(...results);
   230→    if (!data.nextCursor) break;
   231→    cursor = data.nextCursor;
   232→    if (allGames.length >= MAX_GAMES_PER_PLAYER) break;
   233→  }
   234→  return allGames;
   235→}

   237→/** Classifie une game en catégorie. */
   238→function classifyGame(g) {
   239→  const mode = String(g.mode || "").toLowerCase();
   240→  const rt = String(g.rankedType || "").toLowerCase();
   241→  const isTeam =
   242→    mode === "team" ||
   243→    mode.startsWith("2v2") ||
   244→    mode.startsWith("3v3") ||
   245→    mode.startsWith("4v4") ||
   246→    rt === "2v2";
   247→  const isRanked = rt === "1v1" || rt === "2v2" || rt === "ranked";
   248→  if (isTeam) return isRanked ? "teamRanked" : "teamCasual";
   249→  return isRanked ? "ffaRanked" : "ffaCasual";
   250→}

   252→/** Calcule les wins globales + hebdo depuis une liste de games. */
   253→function computeWinsFromGames(games) {
   254→  const global = { ffaCasual: 0, ffaRanked: 0, teamCasual: 0, teamRanked: 0 };
   255→  const weekly = { ffaCasual: 0, ffaRanked: 0, teamCasual: 0, teamRanked: 0 };
   256→  const now = Date.now();
   257→  for (const g of games) {
   258→    if (g.result !== "victory") continue;
   259→    const cat = classifyGame(g);
   260→    global[cat]++;
   261→    if (g.start && now - new Date(g.start).getTime() < WEEKLY_MS) {
   262→      weekly[cat]++;
   263→    }
   264→  }
   265→  return { global, weekly };
   266→}

   268→/** Charge les stats live pour tous les joueurs connectés (avec cache, en parallèle). */
   269→async function loadLiveStats() {
   270→  if (_connectedPlayers.length === 0) {
   271→    _liveFetchDone = true;
   272→    return;
   273→  }
   274→
   275→  // ── 1. Charger le cache localStorage ──
   276→  let cached = {};
   277→  try {
   278→    const raw = localStorage.getItem(LIVE_CACHE_KEY);
   279→    if (raw) {
   280→      const parsed = JSON.parse(raw);
   281→      if (parsed && typeof parsed === "object") cached = parsed;
   282→    }
   283→  } catch (e) { /* ignore */ }
   284→
   285→  // ── 2. Séparer les joueurs en cache (frais) vs à fetcher ──
   286→  const toFetch = [];
   287→  for (const player of _connectedPlayers) {
   288→    const cacheEntry = cached[player.publicId];
   289→    const isFresh = cacheEntry && (Date.now() - cacheEntry.fetchedAt) < LIVE_CACHE_TTL;
   290→    if (isFresh) {
   291→      _liveStats[player.publicId] = cacheEntry;
   292→      _liveFetchProgress++;
   293→    } else {
   294→      toFetch.push(player);
   295→    }
   296→  }
   297→  // Premier rendu avec les données en cache
   298→  if (_liveFetchProgress > 0) mergeAndRender();
   299→
   300→  // ── 3. Fetch tous les joueurs en parallèle (exemption = 8 concurrent) ──
   301→  const fetchOne = async (player) => {
   302→    try {
   303→      const games = await fetchAllPlayerGames(player.publicId);
   304→      const wins = computeWinsFromGames(games);
   305→      const entry = {
   306→        publicId: player.publicId,
   307→        username: player.username,
   308→        games: games.slice(-20), // garder les 20 plus récentes pour le profil
   309→        global: wins.global,
   310→        weekly: wins.weekly,
   311→        totalGames: games.length,
   312→        fetchedAt: Date.now(),
   313→      };
   314→      _liveStats[player.publicId] = entry;
   315→      cached[player.publicId] = entry;
   316→      localStorage.setItem(LIVE_CACHE_KEY, JSON.stringify(cached));
   317→      console.log(`[dashboard] live: ${player.username} (${player.publicId}) → ${games.length} games, global=${JSON.stringify(wins.global)}`);
   318→    } catch (e) {
   319→      console.warn(`[dashboard] live fetch failed for ${player.publicId}:`, e.message);
   320→    }
   321→    _liveFetchProgress++;
   322→    mergeAndRender(); // rendu progressif après chaque joueur
   323→  };
   324→
   325→  // Lancer tous les fetchs en parallèle
   326→  await Promise.all(toFetch.map(fetchOne));
   327→  _liveFetchDone = true;
   328→}

   331→  if (!ts || !lastUpdateEl) return;
   332→  const d = new Date(typeof ts === "number" ? ts : ts);

   340→/* ════════════════════════════════════════════════════════════════
   341→   Merge ranked.json + stats live → _mergedViews { global, weekly }
   342→   ════════════════════════════════════════════════════════════════ */

   344→function pointsFor(p) {
   345→  return (p.ffaCasualWins || 0) * PTS_FFA_CASUAL
   346→       + (p.ffaRankedWins || 0) * PTS_FFA_RANKED
   347→       + (p.teamCasualWins || 0) * PTS_TEAM_CASUAL
   348→       + (p.teamRankedWins || 0) * PTS_TEAM_RANKED;
   349→}
   350→
   351→/**
   352→ * Fusionne ranked.json (career ranked wins) + live API (casual wins)
   353→ * pour les DEUX vues (global + weekly) d'un coup. Les stats live
   354→ * contiennent déjà `global` et `weekly` pour chaque joueur (cf.
   355→ * computeWinsFromGames), donc on parcourt _liveStats deux fois sans
   356→ * re-fetcher.
   357→ */
   358→function buildMergedViews() {
   359→  const buildForMode = (isWeekly) => {
   360→    const byPid = new Map();
   361→
   362→    // 1. ranked.json → ranked wins carrière (GLOBAL uniquement)
   363→    //    En weekly, on ne peut pas déduire les wins hebdo depuis ranked.json
   364→    //    (career total only) → les joueurs ranked-only ont 0 pts en weekly
   365→    if (_rankedData) {
   366→      const getOrCreate = (pid, name) => {
   367→        let e = byPid.get(pid);
   368→        if (!e) {
   369→          e = {
   370→            publicId: pid, username: name || pid, clan: null,
   371→            ffaCasualWins: 0, ffaRankedWins: 0,
   372→            teamCasualWins: 0, teamRankedWins: 0,
   373→            _hasLive: false,
   374→          };
   375→          byPid.set(pid, e);
   376→        }
   377→        return e;
   378→      };
   379→      for (const p of _rankedData["1v1"] || []) {
   380→        const nm = p.username || p.accountUsername || p.public_id;
   381→        const e = getOrCreate(p.public_id, nm);
   382→        if (!isWeekly) e.ffaRankedWins = p.wins || 0;
   383→        if (nm && nm !== p.public_id) e.username = nm;
   384→      }
   385→      for (const p of _rankedData["2v2"] || []) {
   386→        const nm = p.username || p.accountUsername || p.public_id;
   387→        const e = getOrCreate(p.public_id, nm);
   388→        if (!isWeekly) e.teamRankedWins = p.wins || 0;
   389→        if (nm && nm !== p.public_id) e.username = nm;
   390→      }
   391→    }
   392→
   393→    // 2. Stats live → casual wins + ranked wins
   394→    for (const [pid, live] of Object.entries(_liveStats)) {
   395→      let e = byPid.get(pid);
   396→      if (!e) {
   397→        e = {
   398→          publicId: pid, username: live.username || pid, clan: null,
   399→          ffaCasualWins: 0, ffaRankedWins: 0,
   400→          teamCasualWins: 0, teamRankedWins: 0,
   401→          _hasLive: false,
   402→        };
   403→        byPid.set(pid, e);
   404→      }
   405→      const g = isWeekly ? live.weekly : live.global;
   406→      e.ffaCasualWins = g.ffaCasual;
   407→      e.teamCasualWins = g.teamCasual;
   408→      // Ranked wins :
   409→      //   - Global : max(ranked.json career, API live) — ranked.json est plus complet
   410→      //   - Weekly : API live uniquement (ranked.json n'a pas de breakdown hebdo)
   411→      if (isWeekly) {
   412→        e.ffaRankedWins = g.ffaRanked;
   413→        e.teamRankedWins = g.teamRanked;
   414→      } else {
   415→        e.ffaRankedWins = Math.max(e.ffaRankedWins || 0, g.ffaRanked);
   416→        e.teamRankedWins = Math.max(e.teamRankedWins || 0, g.teamRanked);
   417→      }
   418→      e._hasLive = true;
   419→      if (live.username && live.username !== pid) e.username = live.username;
   420→    }
   421→
   422→    // 3. Calcul des points + tri
   423→    const players = [...byPid.values()].map((p) => ({
   424→      ...p,
   425→      points: pointsFor(p),
   426→    }));
   427→    players.sort((a, b) => b.points - a.points);
   428→    return players;
   429→  };
   430→
   431→  return {
   432→    global: buildForMode(false),
   433→    weekly: buildForMode(true),
   434→  };
   435→}

   783→window.startOwnershipVerification = async function () {
   784→  if (!currentUser) {
   785→    showToast("Veuillez vous connecter d'abord.", "warning");
   786→    return;
   787→  }
   788→  const usernameInput = document.getElementById("profile-username");
   789→  const publicIdInput = document.getElementById("profile-public-id");
   790→  const username = (usernameInput?.value || "").trim();
   791→  const publicId = (publicIdInput?.value || "").trim();
   792→
   793→  if (!username || !publicId) {
   794→    showToast("Veuillez remplir tous les champs.", "warning");
   795→    return;
   796→  }
   797→  if (username.length < 2 || username.length > 30) {
   798→    showToast("Le pseudo doit faire entre 2 et 30 caractères.", "warning");
   799→    return;
   800→  }
   801→  if (!/^[A-Za-z0-9]{8}$/.test(publicId)) {
   802→    showToast("Le Public ID doit faire exactement 8 caractères alphanumériques (ex: HabCsQYR).", "warning");
   803→    return;
   804→  }
   805→  if (/[^a-zA-Z0-9_\- ]/.test(username)) {
   806→    showToast("Le pseudo ne peut contenir que des lettres, chiffres, espaces, _ et -.", "warning");
   807→    return;
   808→  }
   809→
   810→  showToast("Vérification du Public ID…", "info", 3000);
   811→  try {
   812→    const playerData = await fetchOpenFront(`/public/player/${encodeURIComponent(publicId)}`);
   813→    if (!playerData || !playerData.publicId) {
   814→      showToast("Public ID introuvable sur OpenFront. Vérifiez votre saisie.", "error");
   815→      return;
   816→    }
   817→  } catch (e) {
   818→    if (e?.isNotFound || e?.status === 404) {
   819→      showToast("Public ID introuvable sur OpenFront. Vérifiez votre saisie.", "error");
   820→      return;
   821→    }
   822→    showToast("Impossible de vérifier le Public ID (API indisponible). Réessayez plus tard.", "error", 6000);
   823→    console.error("[ownership] API check failed:", e);
   824→    return;
   825→  }
   826→
   827→  // Génération du code challenge TFS-XXXX
   828→  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
   829→  _ownershipCode = "TFS-";
   830→  const rand = crypto.getRandomValues(new Uint8Array(4));
   831→  for (let i = 0; i < 4; i++) _ownershipCode += chars[rand[i] % chars.length];
   832→  _ownershipPublicId = publicId;
   833→  _ownershipUsername = username;

   835→  const s1 = document.getElementById("profile-setup-step1");
   836→  const s2 = document.getElementById("profile-setup-step2");

   840→  const exEl = document.getElementById("ownership-example");
   841→  if (codeEl) codeEl.textContent = _ownershipCode;
   842→  if (exEl) exEl.textContent = _ownershipCode + " " + username;
   843→  showToast("Code généré. Suivez les instructions ci-dessous.", "info");
   844→  if (window.hydrateIcons) window.hydrateIcons(document.getElementById("profile-modal"));
   845→};
