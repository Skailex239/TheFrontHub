   128→   128→// When a player subscribes while someone else exclusively holds their bare
   129→   129→// name, the server renames them to TEMPORARY#### and clears their cooldown so
   130→   130→// the rename is free. Detect it to prompt for a real name.
   131→   131→export function isTemporaryUsername(base: string | null | undefined): boolean {
   132→   132→  return typeof base === "string" && /^TEMPORARY\d{4}$/.test(base);
   133→   133→}
   134→   134→
   135→   135→// Whether a pre-rendered account username belongs to a verified player
   136→   136→// (premium/indefinite bare-name claim holder). The server renders everyone
   137→   137→// else as "base.suffix" and bases can never contain dots
   138→   138→// (AccountUsernameSchema), so a dotless name IS the bare display — no
   139→   139→// dedicated flag needed on list endpoints. TEMPORARY#### server renames are
   140→   140→// bare too but aren't a chosen name, so they don't get the badge (matches
   141→   141→// the verified-name play toggle's eligibility rule).
   142→   142→export function isVerifiedUsername(
   143→   143→  username: string | null | undefined,
   144→   144→): boolean {
   145→   145→  return (
   146→   146→    typeof username === "string" &&
   147→   147→    !username.includes(".") &&
   148→   148→    !isTemporaryUsername(username)
   149→   149→  );
   150→   150→}

   666→   666→// https-only URL: a plain URL check would accept javascript:/http: URLs, and these flow
   667→   667→// into href/src, so restrict the scheme (the config is served, but this fails an injected
   668→   668→// entry closed rather than rendering it).
   669→   669→const HttpsUrlSchema = z.url({ protocol: /^https$/ });
   670→   670→
   671→   671→// One verified-live stream in the homepage streaming features. Produced only by the
   672→   672→// API's cron (Twitch Helix; YouTube RSS + videos.list) and served by GET /streams.json.
   673→   673→// `channel` is the login/handle used to derive the watch URL when `url` is absent;
   674→   674→// image/link fields are validated so a malformed entry fails the feed closed.
   675→   675→//
   676→   676→// `startedAt` is required on purpose. It can only come from a live lookup, so a payload
   677→   677→// built from static config cannot satisfy this schema — which means an API still serving
   678→   678→// the older {enabled, channels} shape fails validation and falls back to "show nothing",
   679→   679→// rather than being mistaken for a current one. Liveness is never decided on the client.
   680→   680→export const LiveStreamSchema = z.object({
   681→   681→  platform: z.enum(["twitch", "youtube"]),
   682→   682→  // Login/handle as it appears in the watch URL (twitch.tv/<login>, youtube.com/@handle).
   683→   683→  // One conservative charset for both platforms — no slashes, queries, or whitespace — so
   684→   684→  // an entry can't smuggle path/query segments into the derived URL.
   685→   685→  channel: z
   686→   686→    .string()
   687→   687→    .min(1)
   688→   688→    .max(100)
   689→   689→    .regex(/^@?[A-Za-z0-9._-]+$/, "invalid channel handle"),
   690→   690→  displayName: z.string().min(1).max(100),
   691→   691→  title: z.string().max(200).optional(),
   692→   692→  viewers: z.number().int().nonnegative().default(0),
   693→   693→  avatarUrl: HttpsUrlSchema.optional(),
   694→   694→  url: HttpsUrlSchema.optional(),
   695→   695→  startedAt: z.iso.datetime(),
   696→   696→});
   697→   697→export type LiveStream = z.infer<typeof LiveStreamSchema>;

   699→   699→// The single served feed (GET /streams.json). `featured` is the embed candidate list in
