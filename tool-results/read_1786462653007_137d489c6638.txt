    67→function escapeHtml(s) {
    68→  if (s == null) return "";
    69→  return String(s)
    70→    .replace(/&/g, "&amp;")
    71→    .replace(/</g, "&lt;")
    72→    .replace(/>/g, "&gt;")
    73→    .replace(/"/g, "&quot;")
    74→    .replace(/'/g, "&#39;");
    75→}

    77→function formatPoints(n) {
    78→  return new Intl.NumberFormat("fr-FR").format(n || 0);

    81→function initials(name) {
    82→  if (!name) return "?";
    83→  const clean = name.replace(/^\[[^\]]+\]\s*/, "").trim();
    84→  const parts = clean.split(/[\s_-]+/).filter(Boolean);
    85→  const letters = parts.length >= 2 ? parts[0][0] + parts[1][0] : clean.slice(0, 2);
    86→  return letters.toUpperCase();
    87→}

   115→async function loadData() {
   116→  // Tentative principale : dashboard_ranking.json (cache-bust)
   117→  try {
   118→    const res = await fetch(`dashboard_ranking.json?v=${Date.now()}`, { cache: "no-store" });
   119→    if (res.ok) {
   120→      const json = await res.json();
   121→      if (json && (json.global?.players?.length || json.weekly?.players?.length)) {
   122→        _data = json;
   123→        if (_data.updatedAt) updateLastUpdateLabel(_data.updatedAt);
   124→        return;
   125→      }
   126→    }
   127→  } catch (e) {
   128→    console.warn("[dashboard] dashboard_ranking.json indisponible:", e.message);
   129→  }
   130→
   131→  // Fallback : ranked.json (top 100 classé 1v1 + 2v2)
   132→  _data = null;
   133→  try {
   134→    const res = await fetch("ranked.json", { cache: "no-store" });
   135→    if (res.ok) _fallbackRanked = await res.json();
   136→  } catch (e) {
   137→    console.warn("[dashboard] ranked.json indisponible:", e.message);
   138→  }
   139→  if (_fallbackRanked?.updatedAt) updateLastUpdateLabel(_fallbackRanked.updatedAt);
   140→}

   142→function updateLastUpdateLabel(ts) {
   143→  if (!ts || !lastUpdateEl) return;
   144→  const d = new Date(typeof ts === "number" ? ts : ts);
   145→  if (Number.isNaN(d.getTime())) return;
   146→  lastUpdateEl.textContent = "Mis à jour le " + new Intl.DateTimeFormat("fr-FR", {
   147→    day: "2-digit", month: "2-digit", year: "numeric",
   148→    hour: "2-digit", minute: "2-digit",
   149→  }).format(d);
   150→}

   161→function pointsFor(p) {
   162→  // Le JSON fournit déjà `points`, mais on le recalcule pour être sûr
   163→  // de la cohérence avec le barème affiché.
   164→  return (p.ffaCasualWins || 0) * PTS_FFA_CASUAL
   165→       + (p.ffaRankedWins || 0) * PTS_FFA_RANKED
   166→       + (p.teamCasualWins || 0) * PTS_TEAM_CASUAL
   167→       + (p.teamRankedWins || 0) * PTS_TEAM_RANKED;
   168→}
   169→
   170→function rankedView() {
   171→  // Construit une vue synthétique à partir du fallback ranked.json
   172→  // (casual = 0 partout, ranked wins = wins du leaderboard).
   173→  if (!_fallbackRanked) return null;
   174→  const byPid = new Map();
   175→  const getOrCreate = (pid, name) => {
   176→    let e = byPid.get(pid);
   177→    if (!e) {
   178→      e = {
   179→        publicId: pid, username: name || pid, clan: null,
   180→        ffaCasualWins: 0, ffaRankedWins: 0,
   181→        teamCasualWins: 0, teamRankedWins: 0,
   182→      };
   183→      byPid.set(pid, e);
   184→    }
   185→    return e;
   186→  };
   187→  for (const p of _fallbackRanked["1v1"] || []) {
   188→    const nm = p.username || p.accountUsername || p.public_id;
   189→    const e = getOrCreate(p.public_id, nm);
   190→    e.ffaRankedWins = p.wins || 0;
   191→    if (nm && nm !== p.public_id) e.username = nm;
   192→  }
   193→  for (const p of _fallbackRanked["2v2"] || []) {
   194→    const nm = p.username || p.accountUsername || p.public_id;
   195→    const e = getOrCreate(p.public_id, nm);
   196→    e.teamRankedWins = p.wins || 0;
   197→    if (nm && nm !== p.public_id) e.username = nm;
   198→  }
   199→  const players = [...byPid.values()].map((p) => ({ ...p, points: pointsFor(p) }));
   200→  players.sort((a, b) => b.points - a.points);
   201→  return {
   202→    from: null, to: null,
   203→    gamesScanned: null,
   204→    players,
   205→    _fallback: true,
   206→  };
   207→}

   213→function render() {
   214→  if (!_data && !_fallbackRanked) {
   215→    view.innerHTML = `
   216→      <div class="dash-empty-state">
   217→        <div class="dash-empty-icon"><i data-icon="chart"></i></div>
   218→        <h3>Aucune donnée disponible</h3>
   219→        <p>Synchronisation en cours… Revenez dans quelques minutes.</p>
   220→      </div>`;
   221→    if (window.hydrateIcons) window.hydrateIcons(view);
   222→    return;
   223→  }
   224→
   225→  const isWeekly = _dashMode === "weekly";
   226→  const active = _data ? (isWeekly ? _data.weekly : _data.global) : rankedView();
   227→  if (!active || !active.players || !active.players.length) {
   228→    view.innerHTML = `
   229→      <div class="dash-empty-state">
   230→        <div class="dash-empty-icon"><i data-icon="chart"></i></div>
   231→        <h3>Aucune donnée disponible</h3>
   232→        <p>${isWeekly ? "Aucune partie scannée cette semaine." : "Synchronisation en cours…"} Revenez dans quelques minutes.</p>
   233→      </div>`;
   234→    if (window.hydrateIcons) window.hydrateIcons(view);
   235→    return;
   236→  }
   237→
   238→  // Calcul des points (cohérence) + tri desc
   239→  const players = active.players
   240→    .map((p) => ({ ...p, points: typeof p.points === "number" ? p.points : pointsFor(p) }))
   241→    .sort((a, b) => b.points - a.points);
   242→  // Ajout du rang
   243→  players.forEach((p, i) => { p.rank = i + 1; });
   244→
   245→  const champion = players[0] ?? null;
   246→  const totalPlayers = players.length;
   247→  const topN = players.slice(0, 50);
   248→  const modeLabel = isWeekly ? "Cette semaine" : "Global";
   249→  const fallbackTag = active._fallback
   250→    ? `<span class="dash-fallback-tag">Données classées uniquement (synchronisation casual en cours)</span>`
   251→    : "";
   252→
   253→  view.innerHTML = `
   254→    ${fallbackTag}
   255→
   256→    <div class="dash-controls-row">
   257→      <div class="dash-toggle" role="tablist" aria-label="Période du classement">
   258→        <button class="dash-toggle-btn ${!isWeekly ? "active" : ""}" data-mode="global" role="tab" aria-selected="${!isWeekly}">Global</button>
   259→        <button class="dash-toggle-btn ${isWeekly ? "active" : ""}" data-mode="weekly" role="tab" aria-selected="${isWeekly}">Cette semaine</button>
   260→      </div>
   261→      <div class="dash-scoring-inline">FFA casual +10 · FFA classé +1 · Team casual +5 · Team classé +1</div>
   262→    </div>
   263→
   264→    <div class="dash-grid">
   265→      ${champion ? renderChampion(champion, isWeekly) : ""}
   266→      ${renderRanking(topN, totalPlayers, modeLabel)}
   267→    </div>
   268→
   269→    <div class="dash-scoring-info">
   270→      <svg viewBox="0 0 24 24" width="14" height="14" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><circle cx="12" cy="12" r="10"/><line x1="12" y1="16" x2="12" y2="12"/><line x1="12" y1="8" x2="12.01" y2="8"/></svg>
   271→      <span>Barème : <strong>FFA casual +10</strong> · <strong>FFA classé +1</strong> · <strong>Team casual +5</strong> · <strong>Team classé +1</strong> (le classé rapporte juste 1 pt, pas en plus). ${isWeekly ? "Vue hebdomadaire = parties scannées sur les 7 derniers jours." : "Vue globale = cumul de toutes les parties scannées."}</span>
   272→    </div>
   273→  `;
   274→
   275→  // Hydrate les icônes <i data-icon>
   276→  if (window.hydrateIcons) window.hydrateIcons(view);
   277→
   278→  // Toggle listeners (les boutons sont dans #dashboard-view)
   279→  view.querySelectorAll(".dash-toggle-btn").forEach((btn) => {
   280→    btn.addEventListener("click", () => {
   281→      _dashMode = btn.dataset.mode;
   282→      render();
   283→    });
   284→  });
   285→}

   340→/* ── Ranking list (right column, flex rows) ── */
   341→function renderRanking(topN, totalPlayers, modeLabel) {
   342→  const rows = topN.map((p) => {
   343→    const name = p.username || p.publicId;
   344→    const profileUrl = p.publicId
   345→      ? `profile.html?pid=${encodeURIComponent(p.publicId)}&player=${encodeURIComponent(name)}`
   346→      : `profile.html?player=${encodeURIComponent(name)}`;
   347→    const trophy = p.rank === 1 ? "🏆" : p.rank === 2 ? "🥈" : p.rank === 3 ? "🥉" : null;
   348→    const rankSlot = trophy
   349→      ? `<span class="dash-rank-trophy" aria-hidden="true">${trophy}</span>`
   350→      : `<span class="dash-rank-badge">${p.rank}</span>`;
   351→    return `
   352→      <a class="dash-row dash-row-link" href="${profileUrl}">
   353→        <span class="dash-rank-slot">${rankSlot}</span>
   354→        <span class="dash-player">
   355→          <span class="dash-player-name">${escapeHtml(name)}</span>
   356→          ${p.clan ? `<span class="dash-player-clan">[${escapeHtml(p.clan)}]</span>` : ""}
   357→        </span>
   358→        <span class="dash-score">
   359→          <span class="dash-score-val">${formatPoints(p.points)}</span><span class="dash-score-suffix">pts</span>
   360→        </span>
   361→      </a>`;
   362→  }).join("");
   363→
   364→  return `
   365→    <section class="dash-section dash-ranking-section">
   366→      <div class="dash-section-header">
   367→        <h2 class="dash-section-title">Classement</h2>
   368→        <span class="dash-section-meta">${totalPlayers} joueurs · ${modeLabel}</span>
   369→      </div>
   370→      <div class="dash-list">
   371→        ${rows || `<p class="dash-empty">Aucun joueur classé pour le moment.</p>`}
   372→      </div>
   373→      <a class="dash-more-btn" href="index.html?tab=ranked">Voir tout le classement</a>
   374→    </section>`;
   375→}

   381→function updateAuthUI(user) {
   382→  const loginBtn = document.getElementById("login-btn-main");
   383→  const userContainer = document.getElementById("user-container");
   384→  if (!loginBtn || !userContainer) return;
   385→
   386→  if (!user) {
   387→    loginBtn.style.display = "flex";
   388→    userContainer.style.display = "none";
   389→    userContainer.classList.remove("open");
   390→    return;
   391→  }
   392→
   393→  loginBtn.style.display = "none";
   394→  userContainer.style.display = "block";
   395→
   396→  const name = user.name || "Joueur";
   397→  const publicId = user.publicId || "Non lié";
   398→
   399→  const setText = (id, val) => {
   400→    const el = document.getElementById(id);
   401→    if (el) el.textContent = val;
   402→  };
   403→  setText("user-display-name", name);
   404→  setText("user-public-id-side", publicId !== "Non lié" ? publicId : "En ligne");
   405→  setText("dropdown-username-display", name);
   406→  setText("dropdown-publicid-display", publicId);
   407→
   408→  const avatarEl = document.getElementById("dropdown-avatar");
   409→  const sidebarAvatarEl = document.getElementById("sidebar-avatar");
   410→  const renderAvatar = (el) => {
   411→    if (!el) return;
   412→    if (user.avatar) {
   413→      el.innerHTML = `<img src="${escapeHtml(user.avatar)}" alt="${escapeHtml(name)}" style="width:100%;height:100%;border-radius:50%;object-fit:cover">`;
   414→    } else {
   415→      el.innerHTML = "";
   416→      el.textContent = initials(name);
   417→      el.style.background = "linear-gradient(135deg,var(--accent),var(--accentL))";
   418→      el.style.color = "#fff";
   419→    }
   420→  };
   421→  renderAvatar(avatarEl);
   422→  renderAvatar(sidebarAvatarEl);
   423→}

   425→onAuthStateChanged(auth, async (user) => {
   426→  if (!user) {
   427→    currentUser = null;
   428→    updateAuthUI(null);
   429→    return;
   430→  }
   431→  currentUser = { uid: user.uid, avatar: user.photoURL, email: user.email };
   432→
   433→  // Lecture du profil Firestore
   434→  let profile = null;
   435→  try {
   436→    const snap = await getDoc(doc(db, "users", user.uid));
   437→    if (snap.exists()) profile = snap.data();
   438→  } catch (e) {
   439→    console.warn("[dashboard] Firestore read error:", e.message);
   440→  }
   441→
   442→  if (profile && profile.publicId) {
   443→    currentUser.name = profile.username;
   444→    currentUser.publicId = profile.publicId;
   445→    updateAuthUI(currentUser);
   446→  } else {
   447→    // Premier login sans profil : on affiche le badge + ouvre le setup modal
   448→    currentUser.name = user.displayName || "Joueur";
   449→    updateAuthUI(currentUser);
   450→    // Redirige vers profile.html pour finaliser le setup (le dashboard n'a pas
   451→    // vocation à héberger tout le flow d'ownership verification ici).
   452→    if (profile == null) {
   453→      // Pas de doc Firestore du tout → l'utilisateur n'a jamais finalisé.
   454→      // On l'envoie sur profile.html qui gère le setup.
   455→      // On évite la boucle en ne redirigeant que si l'URL ne contient pas ?setup=1
   456→      const params = new URLSearchParams(window.location.search);
   457→      if (params.get("setup") !== "1") {

   537→/* ── Ownership verification (pour le #profile-modal copié de index.html) ── */
   538→
   539→window.startOwnershipVerification = async function () {
   540→  if (!currentUser) {
   541→    showToast("Veuillez vous connecter d'abord.", "warning");
   542→    return;
   543→  }
   544→  const usernameInput = document.getElementById("profile-username");
   545→  const publicIdInput = document.getElementById("profile-public-id");
   546→  const username = (usernameInput?.value || "").trim();
   547→  const publicId = (publicIdInput?.value || "").trim();
   548→
   549→  if (!username || !publicId) {
   550→    showToast("Veuillez remplir tous les champs.", "warning");
   551→    return;
   552→  }
   553→  if (username.length < 2 || username.length > 30) {
   554→    showToast("Le pseudo doit faire entre 2 et 30 caractères.", "warning");
   555→    return;
   556→  }
   557→  if (!/^[A-Za-z0-9]{8}$/.test(publicId)) {
   558→    showToast("Le Public ID doit faire exactement 8 caractères alphanumériques (ex: HabCsQYR).", "warning");
   559→    return;
   560→  }
   561→  if (/[^a-zA-Z0-9_\- ]/.test(username)) {
   562→    showToast("Le pseudo ne peut contenir que des lettres, chiffres, espaces, _ et -.", "warning");
   563→    return;
   564→  }
   565→
   566→  showToast("Vérification du Public ID…", "info", 3000);
   567→  try {
   568→    const playerData = await fetchOpenFront(`/public/player/${encodeURIComponent(publicId)}`);
   569→    if (!playerData || !playerData.publicId) {
   570→      showToast("Public ID introuvable sur OpenFront. Vérifiez votre saisie.", "error");
   571→      return;
   572→    }
   573→  } catch (e) {
   574→    if (e?.isNotFound || e?.status === 404) {
   575→      showToast("Public ID introuvable sur OpenFront. Vérifiez votre saisie.", "error");
   576→      return;
   577→    }
   578→    showToast("Impossible de vérifier le Public ID (API indisponible). Réessayez plus tard.", "error", 6000);
   579→    console.error("[ownership] API check failed:", e);
   580→    return;
   581→  }
   582→
   583→  // Génération du code challenge TFS-XXXX
   584→  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
   585→  _ownershipCode = "TFS-";
   586→  const rand = crypto.getRandomValues(new Uint8Array(4));
   587→  for (let i = 0; i < 4; i++) _ownershipCode += chars[rand[i] % chars.length];
   588→  _ownershipPublicId = publicId;
   589→  _ownershipUsername = username;
   590→
   591→  const s1 = document.getElementById("profile-setup-step1");
   592→  const s2 = document.getElementById("profile-setup-step2");
   593→  if (s1) s1.style.display = "none";
   594→  if (s2) s2.style.display = "block";
   595→  const codeEl = document.getElementById("ownership-code");
   596→  const exEl = document.getElementById("ownership-example");
   597→  if (codeEl) codeEl.textContent = _ownershipCode;
   598→  if (exEl) exEl.textContent = _ownershipCode + " " + username;
   599→  showToast("Code généré. Suivez les instructions ci-dessous.", "info");
   600→  if (window.hydrateIcons) window.hydrateIcons(document.getElementById("profile-modal"));
   601→};
   602→
   603→window.confirmOwnershipVerification = async function () {
   604→  if (!_ownershipCode || !_ownershipPublicId) return;
   605→  const btn = document.getElementById("confirm-ownership-btn");
   606→  const original = btn?.textContent || "Confirmer";
   607→  if (btn) { btn.disabled = true; btn.textContent = "Vérification…"; }
   608→  try {
   609→    const gamesData = await fetchOpenFront(`/public/player/${encodeURIComponent(_ownershipPublicId)}/games`);
   610→    const games = Array.isArray(gamesData?.results) ? gamesData.results : [];
   611→    const found = games.some((g) => g.username && g.username.includes(_ownershipCode));
   612→    if (!found) {
   613→      showToast("Code non trouvé dans vos parties récentes. Jouez une partie avec le code dans votre pseudo, puis confirmez.", "error", 6000);
   614→      if (btn) { btn.disabled = false; btn.textContent = original; }
   615→      return;
   616→    }
   617→    await saveUserProfile(_ownershipUsername, _ownershipPublicId);
   618→  } catch (e) {
   619→    console.error("[ownership] Confirmation failed:", e);
   620→    showToast("Erreur lors de la vérification. Réessayez.", "error");
   621→    if (btn) { btn.disabled = false; btn.textContent = original; }
   622→  }
   623→};

   635→async function saveUserProfile(username, publicId) {
   636→  if (!currentUser) throw new Error("No authenticated user");
   637→  try {
   638→    const userDocRef = doc(db, "users", currentUser.uid);
   639→    const existing = (await getDoc(userDocRef)).data() || {};
   640→    await setDoc(userDocRef, {
   641→      username,
   642→      publicId,
   643→      email: currentUser.email,
   644→      verified: true,
   645→      verifiedAt: new Date().toISOString(),
   646→      createdAt: existing.createdAt || new Date().toISOString(),
   647→      updatedAt: new Date().toISOString(),
   648→      openFrontSyncPending: true,
   649→    }, { merge: true });
   650→
   651→    currentUser.name = username;
   652→    currentUser.publicId = publicId;
   653→
   654→    const modal = document.getElementById("profile-modal");
   655→    if (modal) modal.classList.remove("active");
   656→    window.cancelOwnershipVerification();
   657→    updateAuthUI(currentUser);
   658→    showToast("Profil vérifié et enregistré avec succès ! Redirection…", "success");
   659→    setTimeout(() => { window.location.href = `profile.html?publicId=${encodeURIComponent(publicId)}&player=${encodeURIComponent(username)}`; }, 800);
   660→  } catch (e) {
   661→    console.error("[dashboard] Save profile error:", e);
   662→    showToast("Erreur lors de la sauvegarde du profil.", "error");
   663→    throw e;
   664→  }
   665→}
